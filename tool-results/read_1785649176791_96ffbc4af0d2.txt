     1→'use client';
     2→
     3→import { useState, useMemo, useRef, useCallback } from 'react';
     4→import {
     5→  BedSingle, BedDouble, ArrowRight, ArrowLeft, CheckCircle,
     6→  CreditCard, Banknote, Wallet, PenLine, Upload,
     7→  Camera, Loader2, X, ImageIcon, ScanLine,
     8→} from 'lucide-react';
     9→import SignatureCanvas from 'react-signature-canvas';
    10→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    11→import { Button } from '@/components/ui/button';
    12→import { Input } from '@/components/ui/input';
    13→import { Label } from '@/components/ui/label';
    14→import { Badge } from '@/components/ui/badge';
    15→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    16→import { Checkbox } from '@/components/ui/checkbox';
    17→import { Separator } from '@/components/ui/separator';
    18→import { useMotelStore } from '@/lib/store';
    19→import { supabase } from '@/lib/supabase';
    20→import { toast } from 'sonner';
    21→import type { RoomType, PaymentMethod } from '@/lib/types';
    22→import IdScanner, { type ScannedIdData } from './id-scanner';
    23→
    24→const STEPS = [
    25→  'Room',
    26→  'Scan ID',
    27→  'Guest Details',
    28→  'Payment',
    29→  'Terms & Sign',
    30→  'Confirm',
    31→];
    32→
    33→export default function CheckIn() {
    34→  const rooms = useMotelStore((s) => s.rooms);
    35→  const addGuest = useMotelStore((s) => s.addGuest);
    36→  const addStay = useMotelStore((s) => s.addStay);
    37→  const addPayment = useMotelStore((s) => s.addPayment);
    38→  const updateRoomStatus = useMotelStore((s) => s.updateRoomStatus);
    39→  const addActivity = useMotelStore((s) => s.addActivity);
    40→  const setActiveTab = useMotelStore((s) => s.setActiveTab);
    41→
    42→  // Step
    43→  const [step, setStep] = useState(0);
    44→
    45→  // Room selection
    46→  const [roomType, setRoomType] = useState<RoomType>('1-bed');
    47→  const [selectedRoomId, setSelectedRoomId] = useState('');
    48→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('card');
    49→  const [termsAccepted, setTermsAccepted] = useState(false);
    50→
    51→  // Guest form fields
    52→  const [firstName, setFirstName] = useState('');
    53→  const [lastName, setLastName] = useState('');
    54→  const [phone, setPhone] = useState('');
    55→  const [email, setEmail] = useState('');
    56→  const [idNumber, setIdNumber] = useState('');
    57→  const [idType, setIdType] = useState('Driver License');
    58→  const [idState, setIdState] = useState('');
    59→  const [dateOfBirth, setDateOfBirth] = useState('');
    60→  const [address, setAddress] = useState('');
    61→  const [gender, setGender] = useState('');
    62→  const [eyeColor, setEyeColor] = useState('');
    63→  const [height, setHeight] = useState('');
    64→  const [checkInDate, setCheckInDate] = useState(() => new Date().toISOString().split('T')[0]);
    65→  const [checkOutDate, setCheckOutDate] = useState(() =>
    66→    new Date(Date.now() + 86400000).toISOString().split('T')[0]
    67→  );
    68→  const [checkInTime, setCheckInTime] = useState('2:00 PM');
    69→  const [checkOutTime, setCheckOutTime] = useState('10:00 AM');
    70→
    71→  // ID photo
    72→  const [idPhotoPreview, setIdPhotoPreview] = useState<string | null>(null);
    73→  const [idPhotoUrl, setIdPhotoUrl] = useState<string>('');
    74→  const fileInputRef = useRef<HTMLInputElement>(null);
    75→
    76→  // ID scan tracking
    77→  const [idScanned, setIdScanned] = useState(false);
    78→  const [scannedIdImage, setScannedIdImage] = useState<string | undefined>();
    79→
    80→  // Signature
    81→  const sigCanvas = useRef<SignatureCanvas>(null);
    82→  const [signatureDataUrl, setSignatureDataUrl] = useState<string>('');
    83→
    84→  const [completedStayId, setCompletedStayId] = useState('');
    85→  const [submitting, setSubmitting] = useState(false);
    86→
    87→  // ── Derived ──
    88→  const availableRooms = useMemo(
    89→    () => rooms.filter((r) => r.status === 'available' && r.type === roomType),
    90→    [rooms, roomType]
    91→  );
    92→  const selectedRoom = rooms.find((r) => r.id === selectedRoomId);
    93→  const rate = selectedRoom?.rate ?? (roomType === '1-bed' ? 65 : 85);
    94→
    95→  // ── Manual file upload handler ──
    96→  const handleFileUpload = useCallback(async (e: React.ChangeEvent<HTMLInputElement>) => {
    97→    const file = e.target.files?.[0];
    98→    if (!file) return;
    99→
   100→    const reader = new FileReader();
   101→    reader.onload = (ev) => setIdPhotoPreview(ev.target?.result as string);
   102→    reader.readAsDataURL(file);
   103→
   104→    if (supabase) {
   105→      try {
   106→        const fileName = `id_admin_${Date.now()}_${file.name}`;
   107→        const { error } = await supabase.storage.from('ids').upload(fileName, file, {
   108→          contentType: file.type,
   109→          upsert: false,
   110→        });
   111→        if (error) {
   112→          toast.error('Failed to upload ID image to storage');
   113→          return;
   114→        }
   115→        const { data } = supabase.storage.from('ids').getPublicUrl(fileName);
   116→        setIdPhotoUrl(data.publicUrl);
   117→        toast.success('ID image uploaded');
   118→      } catch {
   119→        toast.error('Upload failed');
   120→      }
   121→    } else {
   122→      setIdPhotoUrl(reader.result as string);
   123→    }
   124→  }, []);
   125→
   126→  // ── ID Scan Complete Handler ──
   127→  const handleScanComplete = useCallback((data: ScannedIdData, imageBase64?: string) => {
   128→    // Auto-fill all form fields from scanned data
   129→    if (data.firstName) setFirstName(data.firstName);
   130→    if (data.lastName) setLastName(data.lastName);
   131→    if (data.idNumber) setIdNumber(data.idNumber);
   132→    if (data.idType) setIdType(data.idType);
   133→    if (data.issuingState) setIdState(data.issuingState);
   134→    if (data.dateOfBirth) setDateOfBirth(data.dateOfBirth);
   135→    if (data.gender) setGender(data.gender);
   136→    if (data.eyeColor) setEyeColor(data.eyeColor);
   137→    if (data.height) setHeight(data.height);
   138→
   139→    // Build address string
   140→    if (data.address) {
   141→      const parts = [
   142→        data.address.street,
   143→        data.address.city,
   144→        data.address.state,
   145→        data.address.zipCode,
   146→      ].filter(Boolean);
   147→      if (parts.length > 0) setAddress(parts.join(', '));
   148→    }
   149→
   150→    // Store scanned image
   151→    if (imageBase64) {
   152→      setScannedIdImage(imageBase64);
   153→      setIdPhotoPreview(imageBase64);
   154→      setIdPhotoUrl(imageBase64);
   155→    }
   156→
   157→    setIdScanned(true);
   158→    toast.success('ID data extracted — please review and add contact info');
   159→  }, []);
   160→
   161→  // ── Step validation ──
   162→  const canNext = (): boolean => {
   163→    switch (step) {
   164→      case 0: return availableRooms.length > 0;
   165→      case 1: return idScanned && idNumber.length >= 4;
   166→      case 2: return firstName.length >= 2 && lastName.length >= 2 && phone.length >= 10;
   167→      case 3: return true;
   168→      case 4: return termsAccepted;
   169→      case 5: return false;
   170→      default: return false;
   171→    }
   172→  };
   173→
   174→  const handleNext = () => {
   175→    if (step < STEPS.length - 1) setStep(step + 1);
   176→  };
   177→
   178→  const handleBack = () => {
   179→    if (step > 0) setStep(step - 1);
   180→  };
   181→
   182→  const handleSubmit = async () => {
   183→    if (submitting) return;
   184→
   185→    if (!selectedRoomId || !selectedRoom) {
   186→      toast.error('No room selected. Please go back and select a room.');
   187→      return;
   188→    }
   189→
   190→    if (!termsAccepted) {
   191→      toast.error('Please accept the terms and conditions before completing check-in.');
   192→      return;
   193→    }
   194→
   195→    setSubmitting(true);
   196→
   197→    try {
   198→      const guestId = await addGuest({
   199→        firstName,
   200→        lastName,
   201→        phone,
   202→        email,
   203→        idNumber,
   204→        dateOfBirth,
   205→        idPhotoUrl: idPhotoUrl || undefined,
   206→        idType: idType || undefined,
   207→        idState: idState || undefined,
   208→      });
   209→
   210→      const stayId = await addStay({
   211→        guestId,
   212→        roomId: selectedRoomId,
   213→        checkInDate,
   214→        checkInTime,
   215→        checkOutDate,
   216→        checkOutTime,
   217→        rateAmount: rate,
   218→        status: 'active',
   219→        keyDeposit: 10,
   220→        tvRemoteDeposit: 10,
   221→      });
   222→
   223→      await addPayment({
   224→        stayId,
   225→        amount: rate,
   226→        method: paymentMethod,
   227→        description: `Room charge (1 night, ${roomType})`,
   228→      });
   229→
   230→      await addPayment({
   231→        stayId,
   232→        amount: 20,
   233→        method: paymentMethod,
   234→        description: 'Key + TV remote deposit',
   235→      });
   236→
   237→      updateRoomStatus(selectedRoomId, 'occupied');
   238→
   239→      addActivity({
   240→        guest: `${firstName} ${lastName}`,
   241→        action: 'Check-in',
   242→        room: selectedRoom.roomNumber,
   243→        time: new Date().toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: true }),
   244→        status: 'Success',
   245→      });
   246→
   247→      // Save signature to Supabase if connected
   248→      if (supabase && signatureDataUrl) {
   249→        const { error } = await supabase.from('signatures').insert({
   250→          stay_id: stayId,
   251→          guest_id: guestId,
   252→          signature_data_url: signatureDataUrl,
   253→        });
   254→        if (error) console.error('Failed to save signature:', error);
   255→      }
   256→
   257→      setCompletedStayId(stayId);
   258→      setStep(5);
   259→      toast.success('Check-in completed successfully!');
   260→    } catch (err) {
   261→      console.error('Check-in error:', err);
   262→      toast.error('Check-in failed. Please try again.');
   263→    } finally {
   264→      setSubmitting(false);
   265→    }
   266→  };
   267→
   268→  const resetForm = () => {
   269→    setStep(0);
   270→    setRoomType('1-bed');
   271→    setSelectedRoomId('');
   272→    setPaymentMethod('card');
   273→    setTermsAccepted(false);
   274→    setFirstName('');
   275→    setLastName('');
   276→    setPhone('');
   277→    setEmail('');
   278→    setIdNumber('');
   279→    setIdType('Driver License');
   280→    setIdState('');
   281→    setDateOfBirth('');
   282→    setAddress('');
   283→    setGender('');
   284→    setEyeColor('');
   285→    setHeight('');
   286→    setCheckInDate(new Date().toISOString().split('T')[0]);
   287→    setCheckOutDate(new Date(Date.now() + 86400000).toISOString().split('T')[0]);
   288→    setCheckInTime('2:00 PM');
   289→    setCheckOutTime('10:00 AM');
   290→    setIdPhotoPreview(null);
   291→    setIdPhotoUrl('');
   292→    setSignatureDataUrl('');
   293→    setScannedIdImage(undefined);
   294→    setIdScanned(false);
   295→    setCompletedStayId('');
   296→    setSubmitting(false);
   297→  };
   298→
   299→  return (
   300→    <div className="p-4 lg:p-6">
   301→      <div className="mb-6">
   302→        <h2 className="text-2xl font-bold text-foreground">New Check-In</h2>
   303→        <p className="text-muted-foreground mt-1">Register a new guest and assign a room.</p>
   304→      </div>
   305→
   306→      {/* Step Indicator */}
   307→      <div className="flex items-center gap-1 mb-6 overflow-x-auto pb-2">
   308→        {STEPS.map((label, i) => (
   309→          <div key={label} className="flex items-center gap-1 min-w-0">
   310→            <div
   311→              className={`flex items-center gap-1.5 px-2.5 py-1 rounded-full text-xs font-medium whitespace-nowrap ${
   312→                i === step
   313→                  ? 'bg-primary text-primary-foreground'
   314→                  : i < step
   315→                    ? 'bg-green-100 dark:bg-green-900/30 text-green-700 dark:text-green-400'
   316→                    : 'bg-muted text-muted-foreground'
   317→              }`}
   318→            >
   319→              {i < step ? (
   320→                <CheckCircle className="w-3 h-3" />
   321→              ) : (
   322→                <span>{i + 1}</span>
   323→              )}
   324→              <span className="hidden sm:inline">{label}</span>
   325→            </div>
   326→            {i < STEPS.length - 1 && (
   327→              <ArrowRight className="w-3 h-3 text-muted-foreground shrink-0" />
   328→            )}
   329→          </div>
   330→        ))}
   331→      </div>
   332→
   333→      {/* Step Content */}
   334→      <div className="max-w-2xl">
   335→
   336→        {/* ── Step 0: Room Selection ── */}
   337→        {step === 0 && (
   338→          <Card>
   339→            <CardHeader>
   340→              <CardTitle className="text-lg">Select Room Type</CardTitle>
   341→            </CardHeader>
   342→            <CardContent className="space-y-4">
   343→              <div className="grid grid-cols-2 gap-4">
   344→                <button
   345→                  onClick={() => { setRoomType('1-bed'); setSelectedRoomId(''); }}
   346→                  className={`p-4 rounded-lg border-2 transition-all cursor-pointer text-left ${
   347→                    roomType === '1-bed'
   348→                      ? 'border-primary bg-primary/5'
   349→                      : 'border-border hover:border-primary/40'
   350→                  }`}
   351→                >
   352→                  <BedSingle className="w-8 h-8 mb-2 text-muted-foreground" />
   353→                  <p className="font-semibold">1-Bed King</p>
   354→                  <p className="text-sm text-muted-foreground">$65/night</p>
   355→                </button>
   356→                <button
   357→                  onClick={() => { setRoomType('2-bed'); setSelectedRoomId(''); }}
   358→                  className={`p-4 rounded-lg border-2 transition-all cursor-pointer text-left ${
   359→                    roomType === '2-bed'
   360→                      ? 'border-primary bg-primary/5'
   361→                      : 'border-border hover:border-primary/40'
   362→                  }`}
   363→                >
   364→                  <BedDouble className="w-8 h-8 mb-2 text-muted-foreground" />
   365→                  <p className="font-semibold">2-Bed Queen</p>
   366→                  <p className="text-sm text-muted-foreground">$85/night</p>
   367→                </button>
   368→              </div>
   369→
   370→              <div>
   371→                <Label className="mb-2 block">Available Rooms</Label>
   372→                <Select value={selectedRoomId} onValueChange={setSelectedRoomId}>
   373→                  <SelectTrigger>
   374→                    <SelectValue placeholder={`${availableRooms.length} rooms available`} />
   375→                  </SelectTrigger>
   376→                  <SelectContent>
   377→                    {availableRooms.map((r) => (
   378→                      <SelectItem key={r.id} value={r.id}>
   379→                        Room {r.roomNumber} (${r.rate}/night)
   380→                      </SelectItem>
   381→                    ))}
   382→                  </SelectContent>
   383→                </Select>
   384→              </div>
   385→            </CardContent>
   386→          </Card>
   387→        )}
   388→
   389→        {/* ── Step 1: Scan ID ── */}
   390→        {step === 1 && (
   391→          <div className="space-y-4">
   392→            <IdScanner
   393→              onScanComplete={handleScanComplete}
   394→            />
   395→
   396→            {/* Manual ID entry fallback (always visible) */}
   397→            <Card>
   398→              <CardHeader className="pb-3">
   399→                <CardTitle className="text-sm flex items-center gap-2 text-muted-foreground">
   400→                  <PenLine className="w-4 h-4" /> Or enter manually
   401→                </CardTitle>
   402→              </CardHeader>
   403→              <CardContent className="space-y-3">
   404→                <div className="grid grid-cols-2 gap-3">
   405→                  <div>
   406→                    <Label className="mb-1 block text-xs">ID Type</Label>
   407→                    <Select value={idType} onValueChange={setIdType}>
   408→                      <SelectTrigger className="h-9 text-sm"><SelectValue /></SelectTrigger>
   409→                      <SelectContent>
   410→                        <SelectItem value="Driver License">Driver License</SelectItem>
   411→                        <SelectItem value="State ID">State ID</SelectItem>
   412→                        <SelectItem value="Passport">Passport</SelectItem>
   413→                        <SelectItem value="Military ID">Military ID</SelectItem>
   414→                      </SelectContent>
   415→                    </Select>
   416→                  </div>
   417→                  <div>
   418→                    <Label className="mb-1 block text-xs">ID State</Label>
   419→                    <Input value={idState} onChange={(e) => setIdState(e.target.value)} placeholder="CO" className="h-9 text-sm" />
   420→                  </div>
   421→                </div>
   422→                <div>
   423→                  <Label className="mb-1 block text-xs">ID Number *</Label>
   424→                  <Input value={idNumber} onChange={(e) => { setIdNumber(e.target.value); if (e.target.value.length >= 4) setIdScanned(true); }} placeholder="DL-123456" className="h-9 text-sm" />
   425→                </div>
   426→                <div>
   427→                  <Label className="mb-1 block text-xs">Date of Birth *</Label>
   428→                  <Input type="date" value={dateOfBirth} onChange={(e) => setDateOfBirth(e.target.value)} className="h-9 text-sm" />
   429→                </div>
   430→
   431→                {/* Optional file upload */}
   432→                <div>
   433→                  <Label className="mb-1 block text-xs">Upload ID Photo</Label>
   434→                  <input
   435→                    ref={fileInputRef}
   436→                    type="file"
   437→                    accept="image/*"
   438→                    capture="environment"
   439→                    onChange={handleFileUpload}
   440→                    className="hidden"
   441→                  />
   442→                  {idPhotoPreview ? (
   443→                    <div className="relative rounded-lg overflow-hidden border border-border">
   444→                      <img src={idPhotoPreview} alt="ID Preview" className="w-full max-h-32 object-contain bg-muted/30" />
   445→                      <button
   446→                        onClick={() => { setIdPhotoPreview(null); setIdPhotoUrl(''); }}
   447→                        className="absolute top-1 right-1 w-6 h-6 rounded-full bg-black/60 text-white flex items-center justify-center hover:bg-black/80"
   448→                      >
   449→                        <X className="w-3 h-3" />
   450→                      </button>
   451→                    </div>
   452→                  ) : (
   453→                    <button
   454→                      onClick={() => fileInputRef.current?.click()}
   455→                      className="w-full p-4 border-2 border-dashed border-border rounded-lg flex items-center justify-center gap-2 text-muted-foreground hover:border-primary/40 hover:text-foreground transition-colors cursor-pointer text-xs"
   456→                    >
   457→                      <Upload className="w-4 h-4" />
   458→                      <span>Upload ID photo</span>
   459→                    </button>
   460→                  )}
   461→                </div>
   462→              </CardContent>
   463→            </Card>
   464→          </div>
   465→        )}
   466→
   467→        {/* ── Step 2: Guest Details ── */}
   468→        {step === 2 && (
   469→          <Card>
   470→            <CardHeader>
   471→              <CardTitle className="text-lg">Guest Details</CardTitle>
   472→            </CardHeader>
   473→            <CardContent className="space-y-4">
   474→              {/* Scan info banner */}
   475→              {idScanned && (
   476→                <div className="bg-green-50 dark:bg-green-900/20 border border-green-200 dark:border-green-800 rounded-lg p-3 flex items-center gap-2">
   477→                  <CheckCircle className="w-4 h-4 text-green-600 dark:text-green-400 shrink-0" />
   478→                  <p className="text-xs text-green-700 dark:text-green-300">
   479→                    ID data auto-filled from scan. Please verify and add contact info.
   480→                  </p>
   481→                </div>
   482→              )}
   483→
   484→              <div className="grid grid-cols-2 gap-4">
   485→                <div>
   486→                  <Label htmlFor="firstName" className="mb-1.5 block">First Name *</Label>
   487→                  <Input id="firstName" value={firstName} onChange={(e) => setFirstName(e.target.value)} placeholder="John" />
   488→                </div>
   489→                <div>
   490→                  <Label htmlFor="lastName" className="mb-1.5 block">Last Name *</Label>
   491→                  <Input id="lastName" value={lastName} onChange={(e) => setLastName(e.target.value)} placeholder="Doe" />
   492→                </div>
   493→              </div>
   494→              <div>
   495→                <Label htmlFor="phone" className="mb-1.5 block">Phone *</Label>
   496→                <Input id="phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="(720) 555-0100" />
   497→              </div>
   498→              <div>
   499→                <Label htmlFor="email" className="mb-1.5 block">Email</Label>
   500→                <Input id="email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} placeholder="guest@email.com" />
   501→              </div>
   502→
   503→              {/* Auto-filled ID info (read-only display) */}
   504→              <div className="bg-muted/30 rounded-lg p-3 space-y-2">
   505→                <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider">From ID Scan</p>
   506→                <div className="grid grid-cols-2 gap-3">
   507→                  <div>
   508→                    <p className="text-[10px] text-muted-foreground">ID Number</p>
   509→                    <p className="text-sm font-medium font-mono">{idNumber || '—'}</p>
   510→                  </div>
   511→                  <div>
   512→                    <p className="text-[10px] text-muted-foreground">Date of Birth</p>
   513→                    <p className="text-sm font-medium">{dateOfBirth || '—'}</p>
   514→                  </div>
   515→                  <div>
   516→                    <p className="text-[10px] text-muted-foreground">ID Type</p>
   517→                    <p className="text-sm font-medium">{idType}{idState ? ` — ${idState}` : ''}</p>
   518→                  </div>
   519→                  <div>
   520→                    <p className="text-[10px] text-muted-foreground">Gender</p>
   521→                    <p className="text-sm font-medium">{gender || '—'}</p>
   522→                  </div>
   523→                  {address && (
   524→                    <div className="col-span-2">
   525→                      <p className="text-[10px] text-muted-foreground">Address</p>
   526→                      <p className="text-sm font-medium">{address}</p>
   527→                    </div>
   528→                  )}
   529→                  {(eyeColor || height) && (
   530→                    <>
   531→                      {eyeColor && (
   532→                        <div>
   533→                          <p className="text-[10px] text-muted-foreground">Eye Color</p>
   534→                          <p className="text-sm font-medium">{eyeColor}</p>
   535→                        </div>
   536→                      )}
