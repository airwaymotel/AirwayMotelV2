    51→// ── AAMVA PDF417 Parser ────────────────────────────────────────────
    52→
    53→function parseAAMVA(rawText: string): ScannedIdData | null {
    54→  try {
    55→    // AAMVA format starts with @\n followed by ANSI or AAMVA
    56→    if (!rawText.includes('ANSI') && !rawText.includes('AAMVA')) {
    57→      return null;
    58→    }
    59→
    60→    const result: ScannedIdData = {
    61→      firstName: '',
    62→      middleName: '',
    63→      lastName: '',
    64→      fullName: '',
    65→      dateOfBirth: '',
    66→      gender: '',
    67→      address: { street: '', city: '', state: '', zipCode: '' },
    68→      idNumber: '',
    69→      idType: 'Driver License',
    70→      issuingState: '',
    71→      expirationDate: '',
    72→      issueDate: '',
    73→      eyeColor: '',
    74→      hairColor: '',
    75→      height: '',
    76→      weight: '',
    77→      veteran: false,
    78→      organDonor: false,
    79→      realId: false,
    80→    };
    81→
    82→    // Parse AAMVA subfiles
    83→    const lines = rawText.split(/\r?\n/);
    84→
    85→    // Extract issuer ID (first 2 chars after ANSI/AAMVA)
    86→    const headerMatch = rawText.match(/ANSI(\d{6})/);
    87→    if (headerMatch) {
    88→      const issuerId = headerMatch[1].substring(0, 2);
    89→      // Map common issuer IDs to state codes
    90→      const stateMap: Record<string, string> = {
    91→        '08': 'CO', '06': 'CA', '12': 'FL', '17': 'IL',
    92→        '36': 'NY', '42': 'PA', '48': 'TX', '26': 'MI',
    93→        '39': 'OH', '13': 'GA', '27': 'MN', '25': 'MA',
    94→        '34': 'NJ', '04': 'AR', '29': 'MO', '32': 'NV',
    95→        '01': 'AL', '02': 'AK', '05': 'AZ', '09': 'CT',
    96→        '10': 'DE', '11': 'DC', '15': 'HI', '16': 'ID',
    97→        '18': 'IN', '19': 'IA', '20': 'KS', '21': 'KY',
    98→        '22': 'LA', '23': 'ME', '24': 'MD', '28': 'MS',
    99→        '30': 'MT', '31': 'NE', '33': 'NH', '35': 'NM',
   100→        '37': 'NC', '38': 'ND', '40': 'OK', '41': 'OR',
   101→        '44': 'RI', '45': 'SC', '46': 'SD', '47': 'TN',
   102→        '49': 'UT', '50': 'VT', '51': 'VA', '53': 'WA',
   103→        '54': 'WV', '55': 'WI', '56': 'WY',
   104→      };
   105→      result.issuingState = stateMap[issuerId] || issuerId;
   106→    }
   107→
   108→    // Parse each line for AAMVA elements
   109→    for (const line of lines) {
   110→      // DAA - Full Name (Last, First Middle)
   111→      const daaMatch = line.match(/^DAA(.+)/);
   112→      if (daaMatch) {
   113→        const nameParts = daaMatch[1].split(',').map(s => s.trim());
   114→        if (nameParts.length >= 1) result.lastName = nameParts[0];
   115→        if (nameParts.length >= 2) {
   116→          const firstMiddle = nameParts[1].split(' ').map(s => s.trim());
   117→          result.firstName = firstMiddle[0] || '';
   118→          result.middleName = firstMiddle.slice(1).join(' ');
   119→        }
   120→        result.fullName = `${result.firstName} ${result.middleName} ${result.lastName}`.replace(/\s+/g, ' ').trim();
   121→      }
   122→
   123→      // DAC - First Name
   124→      const dacMatch = line.match(/^DAC(.+)/);
   125→      if (dacMatch) result.firstName = dacMatch[1].trim();
   126→
   127→      // DAD - Middle Name
   128→      const dadMatch = line.match(/^DAD(.+)/);
   129→      if (dadMatch) result.middleName = dadMatch[1].trim();
   130→
   131→      // DCB - Last Name
   132→      const dcbMatch = line.match(/^DCB(.+)/);
   133→      if (dcbMatch) result.lastName = dcbMatch[1].trim();
   134→
   135→      // DCS - Last Name
   136→      const dcsMatch = line.match(/^DCS(.+)/);
   137→      if (dcsMatch) result.lastName = dcsMatch[1].trim();
   138→
   139→      // DBB - Date of Birth (YYYYMMDD or MMDDYYYY)
   140→      const dbbMatch = line.match(/^DBB(.+)/);
   141→      if (dbbMatch) {
   142→        const dobRaw = dbbMatch[1].trim();
   143→        if (dobRaw.length === 8) {
   144→          // Could be YYYYMMDD or MMDDYYYY
   145→          if (dobRaw.startsWith('19') || dobRaw.startsWith('20')) {
   146→            result.dateOfBirth = `${dobRaw.slice(0,4)}-${dobRaw.slice(4,6)}-${dobRaw.slice(6,8)}`;
   147→          } else {
   148→            // MMDDYYYY
   149→            result.dateOfBirth = `${dobRaw.slice(4,8)}-${dobRaw.slice(0,2)}-${dobRaw.slice(2,4)}`;
   150→          }
   151→        }
   152→      }
   153→
   154→      // DBC - Gender
   155→      const dbcMatch = line.match(/^DBC(.+)/);
   156→      if (dbcMatch) {
   157→        const g = dbcMatch[1].trim().toUpperCase();
   158→        result.gender = g === '1' || g === 'M' ? 'Male' : g === '2' || g === 'F' ? 'Female' : g;
   159→      }
   160→
   161→      // DAG - Street Address
   162→      const dagMatch = line.match(/^DAG(.+)/);
   163→      if (dagMatch) result.address.street = dagMatch[1].trim();
   164→
   165→      // DAI - City
   166→      const daiMatch = line.match(/^DAI(.+)/);
   167→      if (daiMatch) result.address.city = daiMatch[1].trim();
   168→
   169→      // DAJ - State
   170→      const dajMatch = line.match(/^DAJ(.+)/);
   171→      if (dajMatch) result.address.state = dajMatch[1].trim();
   172→
   173→      // DAK - ZIP + Country
   174→      const dakMatch = line.match(/^DAK(.+)/);
   175→      if (dakMatch) {
   176→        const zipRaw = dakMatch[1].trim();
   177→        result.address.zipCode = zipRaw.substring(0, 5);
   178→      }
   179→
   180→      // DAQ - ID Number
   181→      const daqMatch = line.match(/^DAQ(.+)/);
   182→      if (daqMatch) result.idNumber = daqMatch[1].trim();
   183→
   184→      // DBA - Expiration Date
   185→      const dbaMatch = line.match(/^DBA(.+)/);
   186→      if (dbaMatch) {
   187→        const expRaw = dbaMatch[1].trim();
   188→        if (expRaw.length === 8) {
   189→          if (expRaw.startsWith('20')) {
   190→            result.expirationDate = `${expRaw.slice(0,4)}-${expRaw.slice(4,6)}-${expRaw.slice(6,8)}`;
   191→          } else {
   192→            result.expirationDate = `${expRaw.slice(4,8)}-${expRaw.slice(0,2)}-${expRaw.slice(2,4)}`;
   193→          }
   194→        }
   195→      }
   196→
   197→      // DBD - Issue Date
   198→      const dbdMatch = line.match(/^DBD(.+)/);
   199→      if (dbdMatch) {
   200→        const issRaw = dbdMatch[1].trim();
   201→        if (issRaw.length === 8) {
   202→          if (issRaw.startsWith('20')) {
   203→            result.issueDate = `${issRaw.slice(0,4)}-${issRaw.slice(4,6)}-${issRaw.slice(6,8)}`;
   204→          } else {
   205→            result.issueDate = `${issRaw.slice(4,8)}-${issRaw.slice(0,2)}-${issRaw.slice(2,4)}`;
   206→          }
   207→        }
   208→      }
   209→
   210→      // DAY - Eye Color
   211→      const dayMatch = line.match(/^DAY(.+)/);
   212→      if (dayMatch) {
   213→        const eyeMap: Record<string, string> = {
   214→          'BLK': 'Black', 'BLU': 'Blue', 'BRO': 'Brown', 'GRY': 'Gray',
   215→          'GRN': 'Green', 'HAZ': 'Hazel', 'MAR': 'Maroon', 'DIC': 'Dichromatic',
   216→        };
   217→        result.eyeColor = eyeMap[dayMatch[1].trim().toUpperCase()] || dayMatch[1].trim();
   218→      }
   219→
   220→      // DAZ - Hair Color
   221→      const dazMatch = line.match(/^DAZ(.+)/);
   222→      if (dazMatch) {
   223→        const hairMap: Record<string, string> = {
   224→          'BAL': 'Bald', 'BLK': 'Black', 'BLN': 'Blond', 'BRO': 'Brown',
   225→          'GRY': 'Gray', 'RED': 'Red/Auburn', 'SDY': 'Sandy', 'WHI': 'White',
   226→        };
   227→        result.hairColor = hairMap[dazMatch[1].trim().toUpperCase()] || dazMatch[1].trim();
   228→      }
   229→
   230→      // DAU - Height
   231→      const dauMatch = line.match(/^DAU(.+)/);
   232→      if (dauMatch) result.height = dauMatch[1].trim();
   233→
   234→      // DAW - Weight
   235→      const dawMatch = line.match(/^DAW(.+)/);
   236→      if (dawMatch) result.weight = dawMatch[1].trim();
   237→
   238→      // DCF - Document Discriminator
   239→      // DCK - Inventory Control Number
   240→
   241→      // Check for ID type
   242→      if (line.includes('ID') || line.includes('IDENTIFICATION')) {
   243→        result.idType = 'State ID';
   244→      }
   245→    }
   246→
   247→    // Build full name if not set from DAA
   248→    if (!result.fullName) {
   249→      result.fullName = `${result.firstName} ${result.middleName} ${result.lastName}`.replace(/\s+/g, ' ').trim();
   250→    }
   251→
   252→    // Set issuing state from address state if not found
   253→    if (!result.issuingState && result.address.state) {
   254→      result.issuingState = result.address.state;
   255→    }
   256→
   257→    // Check if we got enough data
   258→    if (!result.lastName && !result.firstName && !result.idNumber) {
   259→      return null;
   260→    }
   261→
   262→    return result;
   263→  } catch {
   264→    return null;
   265→  }
   266→}

   268→// ── Barcode Scanner Component ──────────────────────────────────────
   269→
   270→function BarcodeScanner({ onScanSuccess }: { onScanSuccess: (data: ScannedIdData) => void }) {
   271→  const scannerRef = useRef<HTMLDivElement>(null);
   272→  const html5QrRef = useRef<any>(null);
   273→  const [status, setStatus] = useState<'starting' | 'scanning' | 'error'>('starting');
   274→  const [errorMsg, setErrorMsg] = useState('');
   275→
   276→  useEffect(() => {
   277→    let scanner: any = null;
   278→    let mounted = true;
   279→
   280→    const startScanner = async () => {
   281→      if (!scannerRef.current) return;
   282→
   283→      try {
   284→        const { Html5Qrcode } = await import('html5-qrcode');
   285→        scanner = new Html5Qrcode('id-barcode-scanner');
   286→        html5QrRef.current = scanner;
   287→
   288→        await scanner.start(
   289→          { facingMode: 'environment' },
   290→          {
   291→            fps: 10,
   292→            qrbox: { width: 280, height: 200 },
   293→            formatsToSupport: [
   294→              // @ts-expect-error - PDF417 format constant
   295→              0, // QR_CODE
   296→              // The library auto-detects PDF417
   297→            ],
   298→          },
   299→          (decodedText: string) => {
   300→            // Try to parse as AAMVA
   301→            const parsed = parseAAMVA(decodedText);
   302→            if (parsed) {
   303→              scanner.stop().catch(() => {});
   304→              onScanSuccess(parsed);
   305→            } else {
   306→              // Even if not AAMVA, the barcode might contain useful data
   307→              // Try to use it as an ID number
   308→              if (decodedText.length > 3) {
   309→                const partial: ScannedIdData = {
   310→                  firstName: '',
   311→                  middleName: '',
   312→                  lastName: '',
   313→                  fullName: '',
   314→                  dateOfBirth: '',
   315→                  gender: '',
   316→                  address: { street: '', city: '', state: '', zipCode: '' },
   317→                  idNumber: decodedText,
   318→                  idType: 'State ID',
   319→                  issuingState: '',
   320→                  expirationDate: '',
   321→                  issueDate: '',
   322→                  eyeColor: '',
   323→                  hairColor: '',
   324→                  height: '',
   325→                  weight: '',
   326→                  veteran: false,
   327→                  organDonor: false,
   328→                  realId: false,
   329→                };
   330→                scanner.stop().catch(() => {});
   331→                onScanSuccess(partial);
   332→              }
   333→            }
   334→          },
   335→          () => {
   336→            // Scan failure - ignore, keep scanning
   337→          }
   338→        );
   339→
   340→        if (mounted) setStatus('scanning');
   341→      } catch (err: any) {
   342→        console.error('Scanner error:', err);
   343→        if (mounted) {
   344→          setStatus('error');
   345→          setErrorMsg(err?.message || 'Failed to start camera. Please check permissions.');
   346→        }
   347→      }
   348→    };
   349→
   350→    startScanner();
   351→
   352→    return () => {
   353→      mounted = false;
   354→      if (scanner) {
   355→        scanner.stop().catch(() => {});
   356→        scanner.clear().catch(() => {});
   357→      }
   358→    };
   359→  }, [onScanSuccess]);
   360→
   361→  const handleRetry = async () => {
   362→    setStatus('starting');
   363→    setErrorMsg('');
   364→    if (html5QrRef.current) {
   365→      try {
   366→        await html5QrRef.current.stop();
   367→        html5QrRef.current.clear();
   368→      } catch {}
   369→    }
   370→    // The useEffect will restart
   371→  };
   372→
   373→  return (
   374→    <div className="space-y-4">
   375→      <div className="relative rounded-xl overflow-hidden bg-black aspect-[4/3]">
   376→        <div id="id-barcode-scanner" ref={scannerRef} className="w-full h-full" />
   377→
   378→        {status === 'starting' && (
   379→          <div className="absolute inset-0 flex items-center justify-center bg-black/80">
   380→            <div className="text-center text-white">
   381→              <Loader2 className="w-8 h-8 animate-spin mx-auto mb-2" />
   382→              <p className="text-sm">Starting camera...</p>
   383→            </div>
   384→          </div>
   385→        )}
   386→
   387→        {status === 'scanning' && (
   388→          <div className="absolute inset-0 pointer-events-none">
   389→            <div className="absolute inset-4 border-2 border-white/30 rounded-lg">
   390→              <div className="absolute top-0 left-0 w-6 h-6 border-t-2 border-l-2 border-white rounded-tl-lg" />
   391→              <div className="absolute top-0 right-0 w-6 h-6 border-t-2 border-r-2 border-white rounded-tr-lg" />
   392→              <div className="absolute bottom-0 left-0 w-6 h-6 border-b-2 border-l-2 border-white rounded-bl-lg" />
   393→              <div className="absolute bottom-0 right-0 w-6 h-6 border-b-2 border-r-2 border-white rounded-br-lg" />
   394→            </div>
   395→            <div className="absolute bottom-4 left-0 right-0 text-center">
   396→              <p className="text-white/80 text-xs bg-black/50 inline-block px-3 py-1 rounded-full">
   397→                Hold the back of the ID to the camera
   398→              </p>
   399→            </div>
   400→          </div>
   401→        )}
   402→
   403→        {status === 'error' && (
   404→          <div className="absolute inset-0 flex items-center justify-center bg-black/80">
   405→            <div className="text-center text-white px-4">
   406→              <AlertCircle className="w-10 h-10 mx-auto mb-2 text-red-400" />
   407→              <p className="text-sm mb-3">{errorMsg}</p>
   408→              <Button variant="outline" size="sm" onClick={handleRetry}>
   409→                <RotateCcw className="w-4 h-4 mr-1" /> Retry
   410→              </Button>
   411→            </div>
   412→          </div>
   413→        )}
   414→      </div>
   415→
   416→      <p className="text-xs text-muted-foreground text-center">
   417→        Position the <strong>PDF417 barcode</strong> on the back of the ID within the frame
   418→      </p>
   419→    </div>
   420→  );
   421→}

   423→// ── Vision Scanner Component ───────────────────────────────────────
   424→
   425→function VisionScanner({ onScanSuccess }: { onScanSuccess: (data: ScannedIdData, imageBase64: string) => void }) {
   426→  const [status, setStatus] = useState<'idle' | 'processing' | 'error'>('idle');
   427→  const [preview, setPreview] = useState<string | null>(null);
   428→  const [errorMsg, setErrorMsg] = useState('');
   429→  const fileInputRef = useRef<HTMLInputElement>(null);
   430→  const cameraInputRef = useRef<HTMLInputElement>(null);
   431→
   432→  const processImage = useCallback(async (file: File) => {
   433→    setStatus('processing');
   434→    setErrorMsg('');
   435→
   436→    try {
   437→      // Convert to base64
   438→      const reader = new FileReader();
   439→      const dataUrl = await new Promise<string>((resolve, reject) => {
   440→        reader.onload = () => resolve(reader.result as string);
   441→        reader.onerror = reject;
   442→        reader.readAsDataURL(file);
   443→      });
   444→
   445→      setPreview(dataUrl);
   446→
   447→      // Send to AI Vision API
   448→      const res = await fetch('/api/scan-id', {
   449→        method: 'POST',
   450→        headers: { 'Content-Type': 'application/json' },
   451→        body: JSON.stringify({ imageBase64: dataUrl }),
   452→      });
   453→
   454→      if (!res.ok) {
   455→        const err = await res.json().catch(() => ({}));
   456→        throw new Error(err.error || 'Failed to process ID image');
   457→      }
   458→
   459→      const { data } = await res.json();
   460→      if (!data) {
   461→        throw new Error('No data extracted from ID');
   462→      }
   463→
   464→      onScanSuccess(data, dataUrl);
   465→      setStatus('idle');
   466→    } catch (err: any) {
   467→      console.error('Vision scan error:', err);
   468→      setStatus('error');
   469→      setErrorMsg(err?.message || 'Failed to scan ID. Please try again.');
   470→    }
   471→  }, [onScanSuccess]);
   472→
   473→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   474→    const file = e.target.files?.[0];
   475→    if (file) processImage(file);
   476→  };
   477→
   478→  const handleRetry = () => {
   479→    setStatus('idle');
   480→    setPreview(null);
   481→    setErrorMsg('');
   482→  };
   483→
   484→  return (
   485→    <div className="space-y-4">
   486→      <input
   487→        ref={fileInputRef}
   488→        type="file"
   489→        accept="image/*"
   490→        onChange={handleFileChange}
   491→        className="hidden"
   492→      />
   493→      <input
   494→        ref={cameraInputRef}
   495→        type="file"
   496→        accept="image/*"
   497→        capture="environment"
   498→        onChange={handleFileChange}
   499→        className="hidden"
   500→      />
   501→
   502→      {status === 'idle' && !preview && (
   503→        <div className="space-y-3">
   504→          <div className="grid grid-cols-2 gap-3">
   505→            <button
   506→              onClick={() => cameraInputRef.current?.click()}
   507→              className="p-6 rounded-xl border-2 border-dashed border-border hover:border-primary/40 hover:bg-primary/5 transition-all cursor-pointer flex flex-col items-center gap-2"
   508→            >
   509→              <Camera className="w-8 h-8 text-muted-foreground" />
   510→              <span className="text-sm font-medium">Take Photo</span>
   511→              <span className="text-xs text-muted-foreground">Use camera</span>
   512→            </button>
   513→            <button
   514→              onClick={() => fileInputRef.current?.click()}
   515→              className="p-6 rounded-xl border-2 border-dashed border-border hover:border-primary/40 hover:bg-primary/5 transition-all cursor-pointer flex flex-col items-center gap-2"
   516→            >
   517→              <Upload className="w-8 h-8 text-muted-foreground" />
   518→              <span className="text-sm font-medium">Upload</span>
   519→              <span className="text-xs text-muted-foreground">From gallery</span>
   520→            </button>
   521→          </div>
   522→          <p className="text-xs text-muted-foreground text-center">
   523→            Take a clear photo of the <strong>front of the ID</strong>
   524→          </p>
   525→        </div>
   526→      )}
   527→
   528→      {status === 'processing' && preview && (
   529→        <div className="space-y-3">
   530→          <div className="relative rounded-xl overflow-hidden border border-border">
   531→            <img src={preview} alt="ID Preview" className="w-full max-h-48 object-contain bg-muted/30" />
   532→            <div className="absolute inset-0 bg-black/40 flex items-center justify-center">
   533→              <div className="text-center text-white">
   534→                <Loader2 className="w-8 h-8 animate-spin mx-auto mb-2" />
   535→                <p className="text-sm font-medium">Analyzing ID...</p>
   536→                <p className="text-xs opacity-80">Extracting information with AI</p>
   537→              </div>
   538→            </div>
   539→          </div>
   540→        </div>
   541→      )}
   542→
   543→      {status === 'error' && (
   544→        <div className="space-y-3">
   545→          {preview && (
