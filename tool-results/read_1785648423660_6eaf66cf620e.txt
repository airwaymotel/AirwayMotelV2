    33→

    35→
    36→export default function CheckIn() {
    37→  const rooms = useMotelStore((s) => s.rooms);
    38→  const addGuest = useMotelStore((s) => s.addGuest);
    39→  const addStay = useMotelStore((s) => s.addStay);
    40→  const addPayment = useMotelStore((s) => s.addPayment);
    41→  const updateRoomStatus = useMotelStore((s) => s.updateRoomStatus);
    42→  const addActivity = useMotelStore((s) => s.addActivity);
    43→  const setActiveTab = useMotelStore((s) => s.setActiveTab);
    44→
    45→  // Step & method
    46→  const [step, setStep] = useState(0);
    47→  const [idMethod, setIdMethod] = useState<IdMethod | null>(null);
    48→
    49→  // Room selection
    50→  const [roomType, setRoomType] = useState<RoomType>('1-bed');
    51→  const [selectedRoomId, setSelectedRoomId] = useState('');
    52→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('card');
    53→  const [termsAccepted, setTermsAccepted] = useState(false);
    54→
    55→  // Guest form fields
    56→  const [firstName, setFirstName] = useState('');
    57→  const [lastName, setLastName] = useState('');
    58→  const [phone, setPhone] = useState('');
    59→  const [email, setEmail] = useState('');
    60→  const [idNumber, setIdNumber] = useState('');
    61→  const [idType, setIdType] = useState('Driver License');
    62→  const [idState, setIdState] = useState('');
    63→  const [dateOfBirth, setDateOfBirth] = useState('');
    64→  const [checkInDate, setCheckInDate] = useState(() => new Date().toISOString().split('T')[0]);
    65→  const [checkOutDate, setCheckOutDate] = useState(() =>
    66→    new Date(Date.now() + 86400000).toISOString().split('T')[0]
    67→  );
    68→  const [checkInTime, setCheckInTime] = useState('2:00 PM');
    69→  const [checkOutTime, setCheckOutTime] = useState('10:00 AM');
    70→
    71→  // ID photo (manual upload)
    72→  const [idPhotoPreview, setIdPhotoPreview] = useState<string | null>(null);
    73→  const [idPhotoUrl, setIdPhotoUrl] = useState<string>('');
    74→  const fileInputRef = useRef<HTMLInputElement>(null);
    75→
    76→  // Signature
    77→  const sigCanvas = useRef<SignatureCanvas>(null);
    78→  const [signatureDataUrl, setSignatureDataUrl] = useState<string>('');
    79→
    80→  // QR / Phone scan
    81→  const [scanSessionId] = useState(() => Math.random().toString(36).substring(2, 10) + Date.now().toString(36));
    82→  const [scanReceived, setScanReceived] = useState(false);
    83→  const [scannedImageUrl, setScannedImageUrl] = useState<string>('');
    84→  const [scannedSignatureUrl, setScannedSignatureUrl] = useState<string>('');
    85→  const realtimeChannelRef = useRef<any>(null);
    86→
    87→  const [completedStayId, setCompletedStayId] = useState('');
    88→  const [submitting, setSubmitting] = useState(false);
    89→
    90→  // ── Derived ──
    91→  const availableRooms = useMemo(
    92→    () => rooms.filter((r) => r.status === 'available' && r.type === roomType),
    93→    [rooms, roomType]
    94→  );
    95→  const selectedRoom = rooms.find((r) => r.id === selectedRoomId);
    96→  const rate = selectedRoom?.rate ?? (roomType === '1-bed' ? 65 : 85);
    97→

    99→  const scanUrl = typeof window !== 'undefined'
   100→    ? `${window.location.origin}/scan/${scanSessionId}`
   101→    : '';
   102→

   104→  useEffect(() => {
   105→    if (!supabase || idMethod !== 'phone-scan' || step !== 3) return;

   107→    const channel = supabase.channel(`scan_${scanSessionId}`, {
   108→      config: { broadcast: { self: true } },
   109→    });

   111→    channel.on('broadcast', { event: 'id_scanned' }, (payload) => {
   112→      const { imageUrl, signatureDataUrl: sigUrl } = payload.payload;
   113→      if (imageUrl) {
   114→        setScannedImageUrl(imageUrl);
   115→        setIdPhotoUrl(imageUrl);
   116→      }
   117→      if (sigUrl) {
   118→        setScannedSignatureUrl(sigUrl);
   119→        setSignatureDataUrl(sigUrl);
   120→      }
   121→      setScanReceived(true);
   122→      toast.success('ID scan received from phone!');
   123→    });

   125→    channel.subscribe((status) => {
   126→      if (status === 'SUBSCRIBED') {
   127→        console.log('[Scan] Listening for ID scan on channel:', `scan_${scanSessionId}`);
   128→      }
   129→    });

   131→    realtimeChannelRef.current = channel;
   132→
   133→    return () => {
   134→      if (supabase) supabase.removeChannel(channel);
   135→      realtimeChannelRef.current = null;
   136→    };

   142→    if (!file) return;
   143→
   144→    // Preview
   145→    const reader = new FileReader();
   146→    reader.onload = (ev) => setIdPhotoPreview(ev.target?.result as string);
   147→    reader.readAsDataURL(file);
   148→
   149→    // Upload to Supabase Storage if connected
   150→    if (supabase) {
   151→      try {
   152→        const fileName = `id_admin_${Date.now()}_${file.name}`;
   153→        const { error } = await supabase.storage.from('ids').upload(fileName, file, {
   154→          contentType: file.type,
   155→          upsert: false,
   156→        });
   157→        if (error) {
   158→          toast.error('Failed to upload ID image to storage');
   159→          return;
   160→        }
   161→        const { data } = supabase.storage.from('ids').getPublicUrl(fileName);
   162→        setIdPhotoUrl(data.publicUrl);
   163→        toast.success('ID image uploaded');
   164→      } catch {
   165→        toast.error('Upload failed');
   166→      }
   167→    } else {
   168→      // No Supabase — just use the data URL as preview
   169→      setIdPhotoUrl(reader.result as string);
   170→    }
   171→  }, []);
   172→

   176→      case 0: return availableRooms.length > 0;
   177→      case 1: return idMethod !== null;
   178→      case 2: return firstName.length >= 2 && lastName.length >= 2 && phone.length >= 10;
   179→      case 3:
   180→        if (idMethod === 'phone-scan') return scanReceived;
   181→        return idNumber.length >= 4 && dateOfBirth.length > 0;
   182→      case 4: return true;
   183→      case 5:
   184→        return termsAccepted;
   185→      case 6: return false;
   186→      default: return false;
   187→    }
   188→  };
   189→
   190→  const handleNext = () => {
   191→    if (step < STEPS.length - 1) setStep(step + 1);
   192→  };
   193→
   194→  const handleBack = () => {
   195→    if (step > 0) setStep(step - 1);
   196→  };
   197→
   198→  const handleSubmit = async () => {
   199→    if (submitting) return;
   200→
   201→    if (!selectedRoomId || !selectedRoom) {
   202→      toast.error('No room selected. Please go back and select a room.');
   203→      return;
   204→    }
   205→
   206→    if (!termsAccepted) {
   207→      toast.error('Please accept the terms and conditions before completing check-in.');
   208→      return;

   211→    setSubmitting(true);
   212→
   213→    try {

   215→        firstName,
   216→        lastName,
   217→        phone,
   218→        email,
   219→        idNumber,
   220→        dateOfBirth,
   221→        idPhotoUrl: idPhotoUrl || undefined,
   222→        idType: idType || undefined,
   223→        idState: idState || undefined,
   224→      });
   225→
   226→      const stayId = await addStay({
   227→        guestId,
   228→        roomId: selectedRoomId,
   229→        checkInDate,
   230→        checkInTime,
   231→        checkOutDate,
   232→        checkOutTime,
   233→        rateAmount: rate,
   234→        status: 'active',
   235→        keyDeposit: 10,
   236→        tvRemoteDeposit: 10,
   237→      });

   239→      await addPayment({
   240→        stayId,
   241→        amount: rate,
   242→        method: paymentMethod,
   243→        description: `Room charge (1 night, ${roomType})`,
   244→      });
   245→
   246→      await addPayment({
   247→        stayId,
   248→        amount: 20,
   249→        method: paymentMethod,
   250→        description: 'Key + TV remote deposit',
   251→      });

   253→      updateRoomStatus(selectedRoomId, 'occupied');

   255→      addActivity({
   256→        guest: `${firstName} ${lastName}`,
   257→        action: 'Check-in',
   258→        room: selectedRoom.roomNumber,
   259→        time: new Date().toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: true }),
   260→        status: 'Success',
   261→      });
   262→
   263→      // Save signature to Supabase if connected
   264→      const sigData = idMethod === 'phone-scan' ? scannedSignatureUrl : signatureDataUrl;

   266→        const { error } = await supabase.from('signatures').insert({
   267→          stay_id: stayId,
   268→          guest_id: guestId,
   269→          signature_data_url: sigData,
   270→        });

   272→      }

   274→      setCompletedStayId(stayId);
   275→      setStep(6);
   276→      toast.success('Check-in completed successfully!');
   277→    } catch (err) {
   278→      console.error('Check-in error:', err);
   279→      toast.error('Check-in failed. Please try again.');
   280→    } finally {
   281→      setSubmitting(false);
   282→    }
   283→  };
   284→
   285→  const resetForm = () => {
   286→    setStep(0);
   287→    setIdMethod(null);
   288→    setRoomType('1-bed');
   289→    setSelectedRoomId('');
   290→    setPaymentMethod('card');
   291→    setTermsAccepted(false);
   292→    setFirstName('');
   293→    setLastName('');
   294→    setPhone('');
   295→    setEmail('');
   296→    setIdNumber('');
   297→    setIdType('Driver License');
   298→    setIdState('');
   299→    setDateOfBirth('');
   300→    setCheckInDate(new Date().toISOString().split('T')[0]);
   301→    setCheckOutDate(new Date(Date.now() + 86400000).toISOString().split('T')[0]);
   302→    setCheckInTime('2:00 PM');
   303→    setCheckOutTime('10:00 AM');
   304→    setIdPhotoPreview(null);
   305→    setIdPhotoUrl('');
   306→    setSignatureDataUrl('');
   307→    setScannedImageUrl('');
   308→    setScannedSignatureUrl('');
   309→    setScanReceived(false);
   310→    setCompletedStayId('');
   311→    setSubmitting(false);
   312→  };
   313→
   314→  return (
   315→    <div className="p-4 lg:p-6">
   316→      <div className="mb-6">
   317→        <h2 className="text-2xl font-bold text-foreground">New Check-In</h2>
   318→        <p className="text-muted-foreground mt-1">Register a new guest and assign a room.</p>

   320→
   321→      {/* Step Indicator */}
   322→      <div className="flex items-center gap-1 mb-6 overflow-x-auto pb-2">
   323→        {STEPS.map((label, i) => (
   324→          <div key={label} className="flex items-center gap-1 min-w-0">
   325→            <div

   327→                i === step
   328→                  ? 'bg-primary text-primary-foreground'
   329→                  : i < step
   330→                    ? 'bg-green-100 dark:bg-green-900/30 text-green-700 dark:text-green-400'
   331→                    : 'bg-muted text-muted-foreground'

   333→            >
   334→              {i < step ? (
   335→                <CheckCircle className="w-3 h-3" />

   337→                <span>{i + 1}</span>
   338→              )}
   339→              <span className="hidden sm:inline">{label}</span>

   344→          </div>
   345→        ))}
   346→      </div>
   347→
   348→      {/* Step Content */}

   350→
   351→        {/* ── Step 0: Room Selection ── */}
   352→        {step === 0 && (
   353→          <Card>
   354→            <CardHeader>
   355→              <CardTitle className="text-lg">Select Room Type</CardTitle>
   356→            </CardHeader>
   357→            <CardContent className="space-y-4">
   358→              <div className="grid grid-cols-2 gap-4">
   359→                <button
   360→                  onClick={() => { setRoomType('1-bed'); setSelectedRoomId(''); }}
   361→                  className={`p-4 rounded-lg border-2 transition-all cursor-pointer text-left ${
   362→                    roomType === '1-bed'
   363→                      ? 'border-primary bg-primary/5'
   364→                      : 'border-border hover:border-primary/40'

   366→                >
   367→                  <BedSingle className="w-8 h-8 mb-2 text-muted-foreground" />
   368→                  <p className="font-semibold">1-Bed King</p>

   370→                </button>

   372→                  onClick={() => { setRoomType('2-bed'); setSelectedRoomId(''); }}

   379→                  <BedDouble className="w-8 h-8 mb-2 text-muted-foreground" />

   389→                    <SelectValue placeholder={`${availableRooms.length} rooms available`} />
   390→                  </SelectTrigger>
   391→                  <SelectContent>
   392→                    {availableRooms.map((r) => (
   393→                      <SelectItem key={r.id} value={r.id}>
   394→                        Room {r.roomNumber} (${r.rate}/night)
   395→                      </SelectItem>
   396→                    ))}
   397→                  </SelectContent>
   398→                </Select>
   399→              </div>
   400→            </CardContent>
   401→          </Card>
   402→        )}
   403→
   404→        {/* ── Step 1: ID Method Choice ── */}
   405→        {step === 1 && (
   406→          <Card>
   407→            <CardHeader>
   408→              <CardTitle className="text-lg">How will you verify the guest&apos;s ID?</CardTitle>
   409→            </CardHeader>
   410→            <CardContent className="space-y-4">
   411→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   412→                {/* Manual Entry */}
   413→                <button
   414→                  onClick={() => setIdMethod('manual')}
   415→                  className={`p-6 rounded-lg border-2 transition-all cursor-pointer text-left space-y-3 ${
   416→                    idMethod === 'manual'
   417→                      ? 'border-primary bg-primary/5'
   418→                      : 'border-border hover:border-primary/40'
   419→                  }`}
   420→                >
   421→                  <div className="flex items-center gap-3">
   422→                    <Keyboard className="w-8 h-8 text-muted-foreground" />
   423→                    <span className="font-semibold text-lg">Manual Entry</span>
   424→                  </div>
   425→                  <p className="text-sm text-muted-foreground">
   426→                    Type in ID details and upload a photo of the ID from this computer.
   427→                  </p>
   428→                  {idMethod === 'manual' && (
   429→                    <Badge className="bg-primary text-primary-foreground">Selected</Badge>
   430→                  )}
   431→                </button>
   432→
   433→                {/* Phone Scan */}
   434→                <button
   435→                  onClick={() => setIdMethod('phone-scan')}
   436→                  className={`p-6 rounded-lg border-2 transition-all cursor-pointer text-left space-y-3 ${
   437→                    idMethod === 'phone-scan'
   438→                      ? 'border-primary bg-primary/5'
   439→                      : 'border-border hover:border-primary/40'
   440→                  }`}
   441→                >
   442→                  <div className="flex items-center gap-3">
   443→                    <Smartphone className="w-8 h-8 text-muted-foreground" />
   444→                    <span className="font-semibold text-lg">Phone Scan</span>
   445→                  </div>
   446→                  <p className="text-sm text-muted-foreground">
   447→                    Show a QR code to the guest. They scan it with their phone camera to upload their ID photo and sign.
   448→                  </p>
   449→                  {idMethod === 'phone-scan' && (
   450→                    <Badge className="bg-primary text-primary-foreground">Selected</Badge>
   451→                  )}
   452→                </button>
   453→              </div>
   454→            </CardContent>
   455→          </Card>
   456→        )}
   457→
   458→        {/* ── Step 2: Guest Details ── */}
   459→        {step === 2 && (
   460→          <Card>
   461→            <CardHeader>
   462→              <CardTitle className="text-lg">Guest Details</CardTitle>
   463→            </CardHeader>
   464→            <CardContent className="space-y-4">
   465→              <div className="grid grid-cols-2 gap-4">
   466→                <div>
   467→                  <Label htmlFor="firstName" className="mb-1.5 block">First Name *</Label>
   468→                  <Input id="firstName" value={firstName} onChange={(e) => setFirstName(e.target.value)} placeholder="John" />
   469→                </div>
   470→                <div>
   471→                  <Label htmlFor="lastName" className="mb-1.5 block">Last Name *</Label>
   472→                  <Input id="lastName" value={lastName} onChange={(e) => setLastName(e.target.value)} placeholder="Doe" />
   473→                </div>
   474→              </div>
   475→              <div>
   476→                <Label htmlFor="phone" className="mb-1.5 block">Phone *</Label>
   477→                <Input id="phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="(720) 555-0100" />
   478→              </div>
   479→              <div>
   480→                <Label htmlFor="email" className="mb-1.5 block">Email</Label>

   482→              </div>
   483→              <div className="grid grid-cols-2 gap-4">

   487→                </div>
   488→                <div>
   489→                  <Label className="mb-1.5 block">Check-in Time</Label>

   492→                    <SelectContent>
   493→                      {['12:00 PM', '1:00 PM', '2:00 PM', '3:00 PM', '4:00 PM', '5:00 PM', '6:00 PM'].map((t) => (
   494→                        <SelectItem key={t} value={t}>{t}</SelectItem>
   495→                      ))}
   496→                    </SelectContent>
   497→                  </Select>
   498→                </div>
   499→              </div>
   500→              <div className="grid grid-cols-2 gap-4">
   501→                <div>
   502→                  <Label className="mb-1.5 block">Check-out Date</Label>
   503→                  <Input type="date" value={checkOutDate} onChange={(e) => setCheckOutDate(e.target.value)} />
   504→                </div>
   505→                <div>
   506→                  <Label className="mb-1.5 block">Check-out Time</Label>
   507→                  <Select value={checkOutTime} onValueChange={setCheckOutTime}>
   508→                    <SelectTrigger><SelectValue /></SelectTrigger>
   509→                    <SelectContent>
   510→                      {['10:00 AM', '11:00 AM', '12:00 PM'].map((t) => (
   511→                        <SelectItem key={t} value={t}>{t}</SelectItem>
   512→                      ))}
   513→                    </SelectContent>
   514→                  </Select>
   515→                </div>
   516→              </div>
   517→            </CardContent>
   518→          </Card>

   520→
   521→        {/* ── Step 3: ID Info / Scan ── */}

   529→            <CardContent className="space-y-4">
   530→              <div className="grid grid-cols-2 gap-4">

   533→                  <Select value={idType} onValueChange={setIdType}>
   534→                    <SelectTrigger><SelectValue /></SelectTrigger>
   535→                    <SelectContent>
   536→                      <SelectItem value="Driver License">Driver License</SelectItem>
   537→                      <SelectItem value="State ID">State ID</SelectItem>
   538→                      <SelectItem value="Passport">Passport</SelectItem>
   539→                      <SelectItem value="Military ID">Military ID</SelectItem>
   540→                    </SelectContent>
   541→                  </Select>
   542→                </div>
   543→                <div>
   544→                  <Label className="mb-1.5 block">ID State</Label>
   545→                  <Input value={idState} onChange={(e) => setIdState(e.target.value)} placeholder="CO" />
