     1→import React, { useState, useEffect, useRef } from 'react';
     2→import { BedSingle, BedDouble, Camera, ArrowRight, Info, CheckCircle, Smartphone } from 'lucide-react';
     3→import SectionCard from '../components/ui/SectionCard';
     4→import { supabase } from '../lib/supabase';
     5→import { useNavigate } from 'react-router-dom';
     6→import { useForm } from 'react-hook-form';
     7→import { zodResolver } from '@hookform/resolvers/zod';
     8→import { z } from 'zod';
     9→import SignatureCanvas from 'react-signature-canvas';
    10→import { QRCodeCanvas } from 'qrcode.react';
    11→import { toast } from 'sonner';
    12→import { TERMS_AND_CONDITIONS, TERMS_HEADER } from '../data/termsAndConditions';
    13→
    14→const schema = z.object({
    15→  firstName: z.string().min(2, "First name must be at least 2 characters"),
    16→  lastName: z.string().min(2, "Last name must be at least 2 characters"),
    17→  dob: z.string().min(1, "Date of birth is required"),
    18→  idNumber: z.string().min(4, "ID Number is required"),
    19→  phone: z.string().min(10, "Valid phone number required"),
    20→  email: z.string().email("Invalid email address").or(z.literal('')),
    21→  checkInDate: z.string().min(1, "Required"),
    22→  checkInTime: z.string().min(1, "Required"),
    23→  checkOutDate: z.string().min(1, "Required"),
    24→  checkOutTime: z.string().min(1, "Required"),
    25→});
    26→
    27→export default function CheckInPage() {
    28→  const navigate = useNavigate();
    29→  const sigCanvas = useRef(null);
    30→  
    31→  const [roomType, setRoomType] = useState('1-bed');
    32→  const [paymentMethod, setPaymentMethod] = useState('card');
    33→  const [availableRooms, setAvailableRooms] = useState([]);
    34→  const [selectedRoomId, setSelectedRoomId] = useState('');
    35→  
    36→  const [showQRModal, setShowQRModal] = useState(false);
    37→  const [digitalKeyData, setDigitalKeyData] = useState(null);
    38→  const [assignedRoomNumber, setAssignedRoomNumber] = useState('');
    39→
    40→  const [idImageUrl, setIdImageUrl] = useState(null);
    41→  const [showMobileScanModal, setShowMobileScanModal] = useState(false);
    42→  const [scanSessionId, setScanSessionId] = useState('');
    43→  const [parsedIdData, setParsedIdData] = useState(null);
    44→  const [mobileSignatureDataUrl, setMobileSignatureDataUrl] = useState(null);
    45→
    46→  // Holds the active Realtime channel for the mobile ID-scan handoff so we
    47→  // can remove it on success or when the admin closes the modal.
    48→  const scanChannelRef = useRef(null);
    49→
    50→  const { register, handleSubmit, formState: { errors, isSubmitting }, reset } = useForm({
    51→    resolver: zodResolver(schema),
    52→    defaultValues: {
    53→      checkInDate: new Date().toISOString().split('T')[0],
    54→      checkInTime: new Date().toLocaleTimeString('en-GB', {hour: '2-digit', minute:'2-digit'}),
    55→      checkOutTime: '11:00',
    56→      email: ''
    57→    }
    58→  });
    59→
    60→  const startMobileScanSession = () => {
    61→    const sessionId = Math.random().toString(36).substring(7);
    62→    setScanSessionId(sessionId);
    63→
    64→    // Subscribe BEFORE showing the QR so the channel is listening by the time
    65→    // the mobile device uploads — otherwise a fast upload can beat subscription
    66→    // and the broadcast is missed.
    67→    const channel = supabase.channel(`scan_${sessionId}`)
    68→      .on('broadcast', { event: 'id_scanned' }, (payload) => {
    69→        const { imageUrl, parsedData, signatureDataUrl } = payload.payload;
    70→        setIdImageUrl(imageUrl);
    71→        
    72→        if (parsedData) {
    73→          setParsedIdData(parsedData);
    74→          reset((prev) => ({
    75→            ...prev,
    76→            firstName: parsedData.firstName || prev.firstName,
    77→            lastName: parsedData.lastName || prev.lastName,
    78→            dob: parsedData.dob || prev.dob,
    79→            idNumber: parsedData.idNumber || prev.idNumber,
    80→          }));
    81→        }
    82→
    83→        if (signatureDataUrl) {
    84→          setMobileSignatureDataUrl(signatureDataUrl);
    85→        }
    86→
    87→        if (parsedData && signatureDataUrl) {
    88→          toast.success('ID fields & Signature auto-filled from mobile!');
    89→        } else {
    90→          toast.success('Data successfully received from mobile device!');
    91→        }
    92→        
    93→        setShowMobileScanModal(false);
    94→        if (scanChannelRef.current) {
    95→          supabase.removeChannel(scanChannelRef.current);
    96→          scanChannelRef.current = null;
    97→        }
    98→      })
    99→      .subscribe();
   100→
   101→    scanChannelRef.current = channel;
   102→    setShowMobileScanModal(true);
   103→  };
   104→
   105→  const closeMobileScanModal = () => {
   106→    if (scanChannelRef.current) {
   107→      supabase.removeChannel(scanChannelRef.current);
   108→      scanChannelRef.current = null;
   109→    }
   110→    setShowMobileScanModal(false);
   111→  };
   112→
   113→  useEffect(() => {
   114→    fetchAvailableRooms();
   115→  }, [roomType]);
   116→
   117→  const fetchAvailableRooms = async () => {
   118→    const { data } = await supabase
   119→      .from('rooms')
   120→      .select('*')
   121→      .eq('status', 'available')
   122→      .eq('type', roomType)
   123→      .order('room_number', { ascending: true });
   124→    
   125→    setAvailableRooms(data || []);
   126→    if (data && data.length > 0) {
   127→      setSelectedRoomId(data[0].id);
   128→    } else {
   129→      setSelectedRoomId('');
   130→    }
   131→  };
   132→
   133→  const clearSignature = () => {
   134→    sigCanvas.current?.clear();
   135→  };
   136→
   137→  const onSubmit = async (data) => {
   138→    if (!selectedRoomId) {
   139→      toast.error('Please select an available room.');
   140→      return;
   141→    }
   142→    
   143→    let finalSignatureUrl = mobileSignatureDataUrl;
   144→    if (!finalSignatureUrl) {
   145→      if (!sigCanvas.current || sigCanvas.current.isEmpty()) {
   146→        toast.error('Please provide a digital signature.');
   147→        return;
   148→      }
   149→      finalSignatureUrl = sigCanvas.current.getCanvas().toDataURL('image/png');
   150→    }
   151→
   152→    try {
   153→      // 1. Create Guest
   154→      const { data: guestData, error: guestError } = await supabase
   155→        .from('guests')
   156→        .insert([{
   157→          first_name: data.firstName,
   158→          last_name: data.lastName,
   159→          date_of_birth: data.dob,
   160→          id_number: data.idNumber,
   161→          id_photo_url: idImageUrl || null,
   162→          phone: data.phone,
   163→          email: data.email || null,
   164→          id_type: parsedIdData?.idType || null,
   165→          id_state: parsedIdData?.idState || null,
   166→          address: parsedIdData?.address || null,
   167→          city: parsedIdData?.city || null,
   168→          state: parsedIdData?.state || null,
   169→          zip: parsedIdData?.zip || null,
   170→          sex: parsedIdData?.sex || null,
   171→          eye_color: parsedIdData?.eyeColor || null,
   172→          height: parsedIdData?.height || null
   173→        }])
   174→        .select()
   175→        .single();
   176→        
   177→      if (guestError) throw guestError;
   178→
   179→      // 2. Update Room
   180→      const { error: roomError } = await supabase
   181→        .from('rooms')
   182→        .update({ status: 'occupied' })
   183→        .eq('id', selectedRoomId);
   184→        
   185→      if (roomError) throw roomError;
   186→
   187→      // 3. Create Stay
   188→      const rateAmount = roomType === '1-bed' ? 85 : 120;
   189→      const { data: stayData, error: stayError } = await supabase
   190→        .from('stays')
   191→        .insert([{
   192→          guest_id: guestData.id,
   193→          room_id: selectedRoomId,
   194→          rate_type: 'daily',
   195→          rate_amount: rateAmount,
   196→          check_in_date: data.checkInDate,
   197→          check_in_time: data.checkInTime,
   198→          check_out_date: data.checkOutDate,
   199→          check_out_time: data.checkOutTime,
   200→          status: 'active'
   201→        }])
   202→        .select()
   203→        .single();
   204→        
   205→      if (stayError) throw stayError;
   206→
   207→      // 4. Save Signature
   208→      const { error: sigError } = await supabase
   209→        .from('signatures')
   210→        .insert([{
   211→          stay_id: stayData.id,
   212→          guest_id: guestData.id,
   213→          signature_data_url: finalSignatureUrl
   214→        }]);
   215→        
   216→      if (sigError) throw sigError;
   217→
   218→      // 5. Create Payment
   219→      const totalDue = rateAmount * 1.09;
   220→      const { error: paymentError } = await supabase
   221→        .from('payments')
   222→        .insert([{
   223→          stay_id: stayData.id,
   224→          amount: totalDue,
   225→          method: paymentMethod,
   226→          receipt_generated: true
   227→        }]);
   228→        
   229→      if (paymentError) throw paymentError;
   230→
   231→      toast.success('Check-in completed successfully!');
   232→      
   233→      const roomInfo = availableRooms.find(r => r.id === selectedRoomId);
   234→      setAssignedRoomNumber(roomInfo?.room_number || '');
   235→      setDigitalKeyData(JSON.stringify({
   236→        stay_id: stayData.id,
   237→        room: roomInfo?.room_number,
   238→        token: Math.random().toString(36).substring(7)
   239→      }));
   240→      setShowQRModal(true);
   241→
   242→    } catch (error) {
   243→      console.error('Error during check-in:', error);
   244→      toast.error('Failed to complete check-in: ' + error.message);
   245→    }
   246→  };
   247→
   248→  return (
   249→    <div className="flex-1 p-container-padding max-w-5xl mx-auto w-full">
   250→      <header className="mb-10">
   251→        <h2 className="text-display-lg font-bold text-primary mb-2">New Guest Check-In</h2>
   252→        <p className="text-on-surface-variant text-body-md">Complete all sections to generate the digital key and room assignment.</p>
   253→      </header>
   254→
   255→      <form className="space-y-8 pb-32" onSubmit={handleSubmit(onSubmit)}>
   256→        
   257→        {/* Section 1: Room Selection */}
   258→        <SectionCard 
   259→          number="1" 
   260→          title="Room Selection" 
   261→          rightContent={<span className="text-secondary font-bold text-title-sm">${roomType === '1-bed' ? '85.00' : '120.00'} <span className="font-normal text-on-surface-variant text-body-sm">/ night</span></span>}
   262→        >
   263→          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
   264→            <div className="space-y-2">
   265→              <label className="text-label-caps text-on-surface-variant uppercase font-bold">Room Type</label>
   266→              <div className="grid grid-cols-2 gap-3">
   267→                <button 
   268→                  type="button" 
   269→                  onClick={() => setRoomType('1-bed')}
   270→                  className={`border-2 p-4 rounded-lg flex flex-col items-center gap-2 transition-all cursor-pointer ${
   271→                    roomType === '1-bed' 
   272→                      ? 'border-secondary bg-secondary-fixed text-on-secondary-fixed' 
   273→                      : 'border-outline-variant hover:border-secondary opacity-60'
   274→                  }`}
   275→                >
   276→                  <BedSingle className="w-6 h-6" />
   277→                  <span className="font-bold">1-Bed Standard</span>
   278→                </button>
   279→                <button 
   280→                  type="button" 
   281→                  onClick={() => setRoomType('2-bed')}
   282→                  className={`border-2 p-4 rounded-lg flex flex-col items-center gap-2 transition-all cursor-pointer ${
   283→                    roomType === '2-bed' 
   284→                      ? 'border-secondary bg-secondary-fixed text-on-secondary-fixed' 
   285→                      : 'border-outline-variant hover:border-secondary opacity-60'
   286→                  }`}
   287→                >
   288→                  <BedDouble className="w-6 h-6" />
   289→                  <span className="font-bold">2-Bed Queen</span>
   290→                </button>
   291→              </div>
   292→            </div>
   293→            <div className="space-y-2">
   294→              <label className="text-label-caps text-on-surface-variant uppercase font-bold">Available Room Number</label>
   295→              <select 
   296→                value={selectedRoomId}
   297→                onChange={(e) => setSelectedRoomId(e.target.value)}
   298→                className="w-full h-14 bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary text-body-md"
   299→                required
   300→              >
   301→                <option value="" disabled>Select a room...</option>
   302→                {availableRooms.map(room => (
   303→                  <option key={room.id} value={room.id}>Room {room.room_number} - Floor {room.floor}</option>
   304→                ))}
   305→              </select>
   306→              {availableRooms.length === 0 && (
   307→                <p className="text-error text-sm mt-1">No rooms available for this type.</p>
   308→              )}
   309→            </div>
   310→          </div>
   311→        </SectionCard>
   312→
   313→        {/* Section 2 & 3 */}
   314→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
   315→          <section className="lg:col-span-2 bg-white rounded-xl shadow-sm border border-outline-variant overflow-hidden">
   316→            <div className="bg-surface-container-low px-6 py-4 border-b border-outline-variant flex items-center gap-3">
   317→              <span className="w-8 h-8 rounded-full bg-secondary text-white flex items-center justify-center font-bold text-sm">2</span>
   318→              <h3 className="text-title-sm text-on-surface font-bold">Guest ID Identification</h3>
   319→            </div>
   320→            <div className="p-6 space-y-6">
   321→              
   322→              {idImageUrl ? (
   323→                <div className="border-2 border-outline rounded-xl p-4 text-center bg-surface-container-lowest relative">
   324→                  <img src={idImageUrl} alt="Scanned ID" className="h-40 mx-auto rounded object-contain" />
   325→                  <button type="button" onClick={() => setIdImageUrl(null)} className="absolute top-2 right-2 bg-error text-white px-3 py-1 rounded text-xs font-bold uppercase tracking-wider cursor-pointer">Remove</button>
   326→                  <p className="text-secondary font-bold text-sm mt-2 flex items-center justify-center gap-1"><CheckCircle className="w-4 h-4" /> ID Successfully Captured</p>
   327→                </div>
   328→              ) : (
   329→                <div className="grid grid-cols-2 gap-4">
   330→                  <label className="border-2 border-dashed border-outline rounded-xl p-6 text-center bg-surface-container-lowest hover:bg-surface-container transition-colors cursor-pointer group flex flex-col items-center justify-center">
   331→                    <input 
   332→                      type="file" 
   333→                      accept="image/*" 
   334→                      className="hidden" 
   335→                      onChange={(e) => {
   336→                        const file = e.target.files[0];
   337→                        if (file) {
   338→                          const reader = new FileReader();
   339→                          reader.onload = (e) => setIdImageUrl(e.target.result);
   340→                          reader.readAsDataURL(file);
   341→                        }
   342→                      }}
   343→                    />
   344→                    <Camera className="w-8 h-8 text-outline mb-2 group-hover:text-secondary" />
   345→                    <p className="font-bold text-on-surface text-sm">Upload File</p>
   346→                  </label>
   347→                  <div 
   348→                    onClick={startMobileScanSession}
   349→                    className="border-2 border-dashed border-outline rounded-xl p-6 text-center bg-surface-container-lowest hover:bg-surface-container hover:border-secondary transition-colors cursor-pointer group flex flex-col items-center justify-center"
   350→                  >
   351→                    <Smartphone className="w-8 h-8 text-outline mb-2 group-hover:text-secondary" />
   352→                    <p className="font-bold text-on-surface text-sm group-hover:text-secondary">Use Mobile Camera</p>
   353→                  </div>
   354→                </div>
   355→              )}
   356→
   357→              <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   358→                <div className="space-y-1">
   359→                  <label className="text-label-caps text-on-surface-variant uppercase font-bold">First Name</label>
   360→                  <input {...register('firstName')} type="text" placeholder="John" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4" />
   361→                  {errors.firstName && <p className="text-error text-xs">{errors.firstName.message}</p>}
   362→                </div>
   363→                <div className="space-y-1">
   364→                  <label className="text-label-caps text-on-surface-variant uppercase font-bold">Last Name</label>
   365→                  <input {...register('lastName')} type="text" placeholder="Doe" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4" />
   366→                  {errors.lastName && <p className="text-error text-xs">{errors.lastName.message}</p>}
   367→                </div>
   368→                <div className="space-y-1">
   369→                  <label className="text-label-caps text-on-surface-variant uppercase font-bold">Date of Birth</label>
   370→                  <input {...register('dob')} type="date" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4" />
   371→                  {errors.dob && <p className="text-error text-xs">{errors.dob.message}</p>}
   372→                </div>
   373→                <div className="space-y-1">
   374→                  <label className="text-label-caps text-on-surface-variant uppercase font-bold">ID Document Number</label>
   375→                  <input {...register('idNumber')} type="text" placeholder="E12345678" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4" />
   376→                  {errors.idNumber && <p className="text-error text-xs">{errors.idNumber.message}</p>}
   377→                </div>
   378→              </div>
   379→            </div>
   380→          </section>
   381→
   382→          <section className="bg-white rounded-xl shadow-sm border border-outline-variant overflow-hidden">
   383→            <div className="bg-surface-container-low px-6 py-4 border-b border-outline-variant flex items-center gap-3">
   384→              <span className="w-8 h-8 rounded-full bg-secondary text-white flex items-center justify-center font-bold text-sm">3</span>
   385→              <h3 className="text-title-sm text-on-surface font-bold">Contact</h3>
   386→            </div>
   387→            <div className="p-6 space-y-6">
   388→              <div className="space-y-1">
   389→                <label className="text-label-caps text-on-surface-variant uppercase font-bold">Mobile Phone</label>
   390→                <div className="relative">
   391→                  <input {...register('phone')} type="tel" placeholder="(555) 000-0000" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4" />
   392→                  {errors.phone && <p className="text-error text-xs">{errors.phone.message}</p>}
   393→                </div>
   394→              </div>
   395→              <div className="space-y-1">
   396→                <label className="text-label-caps text-on-surface-variant uppercase font-bold">Email Address (Optional)</label>
   397→                <input {...register('email')} type="email" placeholder="guest@example.com" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4" />
   398→                {errors.email && <p className="text-error text-xs">{errors.email.message}</p>}
   399→              </div>
   400→              <div className="mt-4 p-3 bg-secondary-fixed-dim/20 rounded-lg text-body-sm text-on-secondary-fixed-variant flex items-start gap-2">
   401→                <Info className="w-5 h-5 flex-shrink-0" />
   402→                <span>Guest will receive check-out instructions via SMS.</span>
   403→              </div>
   404→            </div>
   405→          </section>
   406→        </div>
   407→
   408→        {/* Section 4 */}
   409→        <SectionCard number="4" title="Stay Duration">
   410→          <div className="flex flex-col md:flex-row items-center gap-8">
   411→            <div className="flex-1 w-full space-y-2">
   412→              <label className="text-label-caps text-on-surface-variant uppercase font-bold">Check-In</label>
   413→              <div className="flex gap-2">
   414→                <div className="w-2/3">
   415→                  <input {...register('checkInDate')} type="date" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4 h-14" />
   416→                  {errors.checkInDate && <p className="text-error text-xs mt-1">{errors.checkInDate.message}</p>}
   417→                </div>
   418→                <div className="w-1/3">
   419→                  <input {...register('checkInTime')} type="time" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4 h-14" />
   420→                  {errors.checkInTime && <p className="text-error text-xs mt-1">{errors.checkInTime.message}</p>}
   421→                </div>
   422→              </div>
   423→            </div>
   424→            <div className="hidden md:flex flex-col items-center text-outline-variant">
   425→              <ArrowRight className="w-6 h-6" />
   426→              <span className="text-[10px] font-bold uppercase mt-1">Nights</span>
   427→            </div>
   428→            <div className="flex-1 w-full space-y-2">
   429→              <label className="text-label-caps text-on-surface-variant uppercase font-bold">Check-Out</label>
   430→              <div className="flex gap-2">
   431→                <div className="w-2/3">
   432→                  <input {...register('checkOutDate')} type="date" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4 h-14" />
   433→                  {errors.checkOutDate && <p className="text-error text-xs mt-1">{errors.checkOutDate.message}</p>}
   434→                </div>
   435→                <div className="w-1/3">
   436→                  <input {...register('checkOutTime')} type="time" className="w-full bg-surface-container rounded-lg border-none focus:ring-2 focus:ring-secondary py-3 px-4 h-14" />
   437→                  {errors.checkOutTime && <p className="text-error text-xs mt-1">{errors.checkOutTime.message}</p>}
   438→                </div>
   439→              </div>
   440→            </div>
   441→          </div>
   442→        </SectionCard>
   443→
   444→        {/* Section 5 */}
   445→        <SectionCard number="5" title="Agreement & Signature">
   446→          {mobileSignatureDataUrl ? (
   447→            <div className="bg-secondary-fixed/30 border-2 border-secondary/20 rounded-xl p-8 flex flex-col items-center justify-center text-center">
   448→              <CheckCircle className="w-12 h-12 text-secondary mb-3" />
   449→              <h4 className="text-title-md font-bold text-on-surface mb-1">Guest Signed via Mobile</h4>
   450→              <p className="text-on-surface-variant text-body-sm">The digital signature and agreement have been securely captured.</p>
   451→              <button type="button" onClick={() => setMobileSignatureDataUrl(null)} className="mt-4 text-error text-label-sm font-bold uppercase hover:underline cursor-pointer">Clear & Re-sign on Desktop</button>
   452→            </div>
   453→          ) : (
   454→            <div className="space-y-6">
   455→              <div className="bg-surface-container rounded-lg p-4 h-48 overflow-y-auto text-body-sm text-on-surface-variant border border-outline-variant leading-relaxed">
   456→                <h4 className="font-bold text-on-surface mb-2">Terms and Conditions of Stay</h4>
   457→                <p className="mb-3 italic">{TERMS_HEADER}</p>
   458→                <ol className="space-y-2 list-decimal list-inside">
   459→                  {TERMS_AND_CONDITIONS.map(term => (
   460→                    <li key={term.number}>{term.text}</li>
   461→                  ))}
