     1→'use client';
     2→
     3→import { useState } from 'react';
     4→import { useRouter } from 'next/navigation';
     5→import {
     6→  Search, RotateCcw, User, Phone, Mail, Calendar,
     7→  CreditCard, Shield, FileText, ArrowRight, BedSingle, BedDouble,
     8→  ImageIcon, PenLine, Plus, Trash2, Pencil, Check, X, Loader2,
     9→} from 'lucide-react';
    10→import { Input } from '@/components/ui/input';
    11→import { Label } from '@/components/ui/label';
    12→import { Badge } from '@/components/ui/badge';
    13→import { Sheet, SheetContent, SheetHeader, SheetTitle } from '@/components/ui/sheet';
    14→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter } from '@/components/ui/dialog';
    15→import { Button } from '@/components/ui/button';
    16→import { Separator } from '@/components/ui/separator';
    17→import { Avatar, AvatarFallback } from '@/components/ui/avatar';
    18→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
    19→import { useMotelStore } from '@/lib/store';
    20→import { toast } from 'sonner';
    21→import RoomCard from './room-card';
    22→import type { Room, RoomType, RoomStatus } from '@/lib/types';
    23→
    24→const FILTERS = ['All', 'Available', 'Occupied', 'Needs Attention'] as const;
    25→
    26→const STATUS_OPTIONS: { value: RoomStatus; label: string; color: string }[] = [
    27→  { value: 'available', label: 'Available', color: 'bg-green-500' },
    28→  { value: 'occupied', label: 'Occupied', color: 'bg-red-500' },
    29→  { value: 'maintenance', label: 'Maintenance', color: 'bg-amber-500' },
    30→  { value: 'cleaning', label: 'Cleaning', color: 'bg-blue-500' },
    31→  { value: 'reserved', label: 'Reserved', color: 'bg-purple-500' },
    32→];
    33→
    34→export default function Rooms() {
    35→  const router = useRouter();
    36→  const rooms = useMotelStore((s) => s.rooms);
    37→  const stays = useMotelStore((s) => s.stays);
    38→  const guests = useMotelStore((s) => s.guests);
    39→  const payments = useMotelStore((s) => s.payments);
    40→  const addRoom = useMotelStore((s) => s.addRoom);
    41→  const updateRoomStatus = useMotelStore((s) => s.updateRoomStatus);
    42→  const updateRoom = useMotelStore((s) => s.updateRoom);
    43→  const deleteRoom = useMotelStore((s) => s.deleteRoom);
    44→
    45→  const [filter, setFilter] = useState<string>('All');
    46→  const [searchQuery, setSearchQuery] = useState('');
    47→  const [selectedRoom, setSelectedRoom] = useState<Room | null>(null);
    48→
    49→  // Add Room dialog state
    50→  const [showAddDialog, setShowAddDialog] = useState(false);
    51→  const [newRoomNumber, setNewRoomNumber] = useState('');
    52→  const [newRoomType, setNewRoomType] = useState<RoomType>('1-bed');
    53→  const [newRoomFloor, setNewRoomFloor] = useState('1');
    54→  const [newRoomStatus, setNewRoomStatus] = useState<RoomStatus>('available');
    55→  const [adding, setAdding] = useState(false);
    56→
    57→  // Edit mode in sheet
    58→  const [editing, setEditing] = useState(false);
    59→  const [editRoomNumber, setEditRoomNumber] = useState('');
    60→  const [editRoomType, setEditRoomType] = useState<RoomType>('1-bed');
    61→  const [editRoomFloor, setEditRoomFloor] = useState('1');
    62→
    63→  // Delete confirmation
    64→  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false);
    65→  const [deleting, setDeleting] = useState(false);
    66→
    67→  const filteredRooms = rooms.filter((room) => {
    68→    if (filter !== 'All') {
    69→      if (filter === 'Available' && room.status !== 'available') return false;
    70→      if (filter === 'Occupied' && room.status !== 'occupied') return false;
    71→      if (filter === 'Needs Attention' && room.status !== 'cleaning' && room.status !== 'maintenance') return false;
    72→    }
    73→    if (searchQuery && !room.roomNumber.includes(searchQuery)) return false;
    74→    return true;
    75→  });
    76→
    77→  const getActiveStayForRoom = (roomId: string) => {
    78→    const stay = stays.find((s) => s.roomId === roomId && s.status === 'active');
    79→    if (!stay) return undefined;
    80→    const guest = guests.find((g) => g.id === stay.guestId);
    81→    if (!guest) return undefined;
    82→    const stayPayments = payments.filter((p) => p.stayId === stay.id);
    83→    return { ...stay, guest, payments: stayPayments };
    84→  };
    85→
    86→  const statusColors: Record<Room['status'], string> = {
    87→    available: 'bg-green-500',
    88→    occupied: 'bg-red-500',
    89→    maintenance: 'bg-amber-500',
    90→    cleaning: 'bg-blue-500',
    91→    reserved: 'bg-purple-500',
    92→  };
    93→
    94→  const activeStay = selectedRoom ? getActiveStayForRoom(selectedRoom.id) : undefined;
    95→
    96→  // ── Add Room Handler ──
    97→  const handleAddRoom = async () => {
    98→    if (!newRoomNumber.trim()) {
    99→      toast.error('Room number is required');
   100→      return;
   101→    }
   102→    if (rooms.some((r) => r.roomNumber === newRoomNumber.trim())) {
   103→      toast.error(`Room ${newRoomNumber} already exists`);
   104→      return;
   105→    }
   106→    setAdding(true);
   107→    try {
   108→      const rate = newRoomType === '1-bed' ? 65 : 85;
   109→      await addRoom({
   110→        roomNumber: newRoomNumber.trim(),
   111→        type: newRoomType,
   112→        floor: parseInt(newRoomFloor) || 1,
   113→        rate,
   114→        status: newRoomStatus,
   115→      });
   116→      toast.success(`Room ${newRoomNumber} added successfully`);
   117→      setShowAddDialog(false);
   118→      setNewRoomNumber('');
   119→      setNewRoomType('1-bed');
   120→      setNewRoomFloor('1');
   121→      setNewRoomStatus('available');
   122→    } catch (err) {
   123→      toast.error('Failed to add room');
   124→    } finally {
   125→      setAdding(false);
   126→    }
   127→  };
   128→
   129→  // ── Delete Room Handler ──
   130→  const handleDeleteRoom = async () => {
   131→    if (!selectedRoom) return;
   132→    if (selectedRoom.status === 'occupied') {
   133→      toast.error('Cannot delete an occupied room. Check out the guest first.');
   134→      setShowDeleteConfirm(false);
   135→      return;
   136→    }
   137→    setDeleting(true);
   138→    try {
   139→      await deleteRoom(selectedRoom.id);
   140→      toast.success(`Room ${selectedRoom.roomNumber} deleted`);
   141→      setSelectedRoom(null);
   142→      setShowDeleteConfirm(false);
   143→    } catch (err) {
   144→      toast.error('Failed to delete room');
   145→    } finally {
   146→      setDeleting(false);
   147→    }
   148→  };
   149→
   150→  // ── Edit Room Handler ──
   151→  const handleSaveEdit = () => {
   152→    if (!selectedRoom) return;
   153→    if (!editRoomNumber.trim()) {
   154→      toast.error('Room number is required');
   155→      return;
   156→    }
   157→    const updates: Partial<Omit<Room, 'id'>> = {};
   158→    if (editRoomNumber !== selectedRoom.roomNumber) updates.roomNumber = editRoomNumber.trim();
   159→    if (editRoomType !== selectedRoom.type) {
   160→      updates.type = editRoomType;
   161→      updates.rate = editRoomType === '1-bed' ? 65 : 85;
   162→    }
   163→    if (parseInt(editRoomFloor) !== selectedRoom.floor) updates.floor = parseInt(editRoomFloor) || 1;
   164→
   165→    if (Object.keys(updates).length > 0) {
   166→      updateRoom(selectedRoom.id, updates);
   167→      setSelectedRoom({ ...selectedRoom, ...updates });
   168→      toast.success('Room updated');
   169→    }
   170→    setEditing(false);
   171→  };
   172→
   173→  const startEditing = () => {
   174→    if (!selectedRoom) return;
   175→    setEditRoomNumber(selectedRoom.roomNumber);
   176→    setEditRoomType(selectedRoom.type);
   177→    setEditRoomFloor(String(selectedRoom.floor));
   178→    setEditing(true);
   179→  };
   180→
   181→  return (
   182→    <div className="p-4 lg:p-6">
   183→      <div className="flex flex-col md:flex-row md:items-end justify-between gap-4 mb-6">
   184→        <div>
   185→          <h2 className="text-2xl font-bold text-foreground">Room Status Board</h2>
   186→          <p className="text-muted-foreground mt-1">Real-time overview of all property units.</p>
   187→        </div>
   188→
   189→        <div className="flex items-center gap-2 flex-wrap">
   190→          <div className="relative">
   191→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   192→            <Input
   193→              placeholder="Room #"
   194→              value={searchQuery}
   195→              onChange={(e) => setSearchQuery(e.target.value)}
   196→              className="pl-8 w-28"
   197→            />
   198→          </div>
   199→
   200→          <div className="flex bg-muted rounded-md p-0.5">
   201→            {FILTERS.map((f) => (
   202→              <button
   203→                key={f}
   204→                onClick={() => setFilter(f)}
   205→                className={`px-3 py-1.5 rounded-sm text-xs font-medium transition-all cursor-pointer ${
   206→                  filter === f
   207→                    ? 'bg-background text-foreground shadow-sm'
   208→                    : 'text-muted-foreground hover:text-foreground'
   209→                }`}
   210→              >
   211→                {f}
   212→              </button>
   213→            ))}
   214→          </div>
   215→
   216→          <Button variant="outline" size="icon" className="h-9 w-9" onClick={() => useMotelStore.getState().loadFromSupabase()}>
   217→            <RotateCcw className="w-4 h-4" />
   218→          </Button>
   219→
   220→          {/* Add Room Button */}
   221→          <Button size="sm" onClick={() => setShowAddDialog(true)} className="h-9">
   222→            <Plus className="w-4 h-4 mr-1.5" />
   223→            Add Room
   224→          </Button>
   225→        </div>
   226→      </div>
   227→
   228→      {/* Room Grid */}
   229→      {filteredRooms.length === 0 ? (
   230→        <div className="text-center py-20 text-muted-foreground bg-card rounded-xl border border-dashed border-border">
   231→          No rooms match your current filters.
   232→        </div>
   233→      ) : (
   234→        <div className="space-y-6">
   235→          {/* Legend */}
   236→          <div className="flex flex-wrap gap-3 text-xs text-muted-foreground">
   237→            {STATUS_OPTIONS.map((s) => (
   238→              <div key={s.value} className="flex items-center gap-1.5">
   239→                <span className={`w-2.5 h-2.5 rounded-full ${s.color}`} />
   240→                <span>{s.label}</span>
   241→              </div>
   242→            ))}
   243→          </div>
   244→
   245→          <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-6 gap-3">
   246→            {filteredRooms.map((room) => (
   247→              <RoomCard
   248→                key={room.id}
   249→                room={room}
   250→                activeStay={getActiveStayForRoom(room.id)}
   251→                onClick={() => { setSelectedRoom(room); setEditing(false); }}
   252→              />
   253→            ))}
   254→          </div>
   255→        </div>
   256→      )}
   257→
   258→      {/* ── Add Room Dialog ── */}
   259→      <Dialog open={showAddDialog} onOpenChange={setShowAddDialog}>
   260→        <DialogContent>
   261→          <DialogHeader>
   262→            <DialogTitle>Add New Room</DialogTitle>
   263→          </DialogHeader>
   264→          <div className="space-y-4 py-2">
   265→            <div className="grid grid-cols-2 gap-4">
   266→              <div className="space-y-2">
   267→                <Label htmlFor="room-number" className="text-xs">Room Number *</Label>
   268→                <Input
   269→                  id="room-number"
   270→                  placeholder="e.g. 15"
   271→                  value={newRoomNumber}
   272→                  onChange={(e) => setNewRoomNumber(e.target.value)}
   273→                />
   274→              </div>
   275→              <div className="space-y-2">
   276→                <Label htmlFor="room-floor" className="text-xs">Floor</Label>
   277→                <Input
   278→                  id="room-floor"
   279→                  type="number"
   280→                  min="1"
   281→                  value={newRoomFloor}
   282→                  onChange={(e) => setNewRoomFloor(e.target.value)}
   283→                />
   284→              </div>
   285→            </div>
   286→            <div className="grid grid-cols-2 gap-4">
   287→              <div className="space-y-2">
   288→                <Label className="text-xs">Room Type</Label>
   289→                <Select value={newRoomType} onValueChange={(v) => setNewRoomType(v as RoomType)}>
   290→                  <SelectTrigger>
   291→                    <SelectValue />
   292→                  </SelectTrigger>
   293→                  <SelectContent>
   294→                    <SelectItem value="1-bed">1-Bed King ($65)</SelectItem>
   295→                    <SelectItem value="2-bed">2-Bed Queen ($85)</SelectItem>
   296→                  </SelectContent>
   297→                </Select>
   298→              </div>
   299→              <div className="space-y-2">
   300→                <Label className="text-xs">Initial Status</Label>
   301→                <Select value={newRoomStatus} onValueChange={(v) => setNewRoomStatus(v as RoomStatus)}>
   302→                  <SelectTrigger>
   303→                    <SelectValue />
   304→                  </SelectTrigger>
   305→                  <SelectContent>
   306→                    {STATUS_OPTIONS.map((s) => (
   307→                      <SelectItem key={s.value} value={s.value}>
   308→                        <span className="flex items-center gap-2">
   309→                          <span className={`w-2 h-2 rounded-full ${s.color}`} />
   310→                          {s.label}
   311→                        </span>
   312→                      </SelectItem>
   313→                    ))}
   314→                  </SelectContent>
   315→                </Select>
   316→              </div>
   317→            </div>
   318→          </div>
   319→          <DialogFooter>
   320→            <Button variant="outline" onClick={() => setShowAddDialog(false)} disabled={adding}>
   321→              Cancel
   322→            </Button>
   323→            <Button onClick={handleAddRoom} disabled={adding || !newRoomNumber.trim()}>
   324→              {adding ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Plus className="w-4 h-4 mr-2" />}
   325→              {adding ? 'Adding...' : 'Add Room'}
   326→            </Button>
   327→          </DialogFooter>
   328→        </DialogContent>
   329→      </Dialog>
   330→
   331→      {/* ── Delete Confirmation Dialog ── */}
   332→      <Dialog open={showDeleteConfirm} onOpenChange={setShowDeleteConfirm}>
   333→        <DialogContent>
   334→          <DialogHeader>
   335→            <DialogTitle>Delete Room {selectedRoom?.roomNumber}?</DialogTitle>
   336→          </DialogHeader>
   337→          <p className="text-sm text-muted-foreground py-2">
   338→            This will permanently remove this room from the system. This action cannot be undone.
   339→          </p>
   340→          <DialogFooter>
   341→            <Button variant="outline" onClick={() => setShowDeleteConfirm(false)} disabled={deleting}>
   342→              Cancel
   343→            </Button>
   344→            <Button variant="destructive" onClick={handleDeleteRoom} disabled={deleting}>
   345→              {deleting ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Trash2 className="w-4 h-4 mr-2" />}
   346→              {deleting ? 'Deleting...' : 'Delete Room'}
   347→            </Button>
   348→          </DialogFooter>
   349→        </DialogContent>
   350→      </Dialog>
   351→
   352→      {/* ── Room Detail Sheet ── */}
   353→      <Sheet open={!!selectedRoom} onOpenChange={(open) => { if (!open) { setSelectedRoom(null); setEditing(false); } }}>
   354→        <SheetContent className="overflow-y-auto">
   355→          {selectedRoom && (
   356→            <>
   357→              <SheetHeader>
   358→                <SheetTitle className="flex items-center gap-3">
   359→                  <span className={`w-3 h-3 rounded-full ${statusColors[selectedRoom.status]}`} />
   360→                  {editing ? `Edit Room ${selectedRoom.roomNumber}` : `Room ${selectedRoom.roomNumber}`}
   361→                </SheetTitle>
   362→              </SheetHeader>
   363→
   364→              {/* ── EDIT MODE ── */}
   365→              {editing ? (
   366→                <div className="mt-8 space-y-5">
   367→                  <div className="space-y-4">
   368→                    <div className="space-y-2">
   369→                      <Label className="text-xs">Room Number</Label>
   370→                      <Input value={editRoomNumber} onChange={(e) => setEditRoomNumber(e.target.value)} />
   371→                    </div>
   372→                    <div className="grid grid-cols-2 gap-4">
   373→                      <div className="space-y-2">
   374→                        <Label className="text-xs">Type</Label>
   375→                        <Select value={editRoomType} onValueChange={(v) => setEditRoomType(v as RoomType)}>
   376→                          <SelectTrigger><SelectValue /></SelectTrigger>
   377→                          <SelectContent>
   378→                            <SelectItem value="1-bed">1-Bed King ($65)</SelectItem>
   379→                            <SelectItem value="2-bed">2-Bed Queen ($85)</SelectItem>
   380→                          </SelectContent>
   381→                        </Select>
   382→                      </div>
   383→                      <div className="space-y-2">
   384→                        <Label className="text-xs">Floor</Label>
   385→                        <Input type="number" min="1" value={editRoomFloor} onChange={(e) => setEditRoomFloor(e.target.value)} />
   386→                      </div>
   387→                    </div>
   388→                  </div>
   389→
   390→                  <Separator />
   391→
   392→                  <div className="flex gap-3">
   393→                    <Button variant="outline" className="flex-1" onClick={() => setEditing(false)}>
   394→                      <X className="w-4 h-4 mr-1.5" /> Cancel
   395→                    </Button>
   396→                    <Button className="flex-1" onClick={handleSaveEdit}>
   397→                      <Check className="w-4 h-4 mr-1.5" /> Save
   398→                    </Button>
   399→                  </div>
   400→                </div>
   401→              ) : (
   402→                <>
   403→                  {/* ── OCCUPIED ROOM: Rich guest detail ── */}
   404→                  {selectedRoom.status === 'occupied' && activeStay ? (
   405→                    <div className="mt-8 space-y-6">
   406→                      {/* Room basics */}
   407→                      <div className="grid grid-cols-2 gap-4">
   408→                        <div>
   409→                          <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider mb-1.5">Type</p>
   410→                          <p className="text-sm font-medium flex items-center gap-1.5">
   411→                            {selectedRoom.type === '1-bed' ? <BedSingle className="w-4 h-4" /> : <BedDouble className="w-4 h-4" />}
   412→                            {selectedRoom.type === '1-bed' ? '1-Bed King' : '2-Bed Queen'}
   413→                          </p>
   414→                        </div>
   415→                        <div>
   416→                          <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider mb-1.5">Rate</p>
   417→                          <p className="text-sm font-medium">${selectedRoom.rate}/night</p>
   418→                        </div>
   419→                        <div>
   420→                          <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider mb-1.5">Floor</p>
   421→                          <p className="text-sm font-medium">{selectedRoom.floor}</p>
   422→                        </div>
   423→                        <div>
   424→                          <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider mb-1.5">Status</p>
   425→                          <Badge variant="destructive" className="text-[10px] uppercase">Occupied</Badge>
   426→                        </div>
   427→                      </div>
   428→
   429→                      <Separator className="my-2" />
   430→
   431→                      {/* Guest Profile */}
   432→                      <div>
   433→                        <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider mb-3 flex items-center gap-1.5">
   434→                          <User className="w-3.5 h-3.5" /> Current Guest
   435→                        </p>
   436→                        <div className="flex items-center gap-4">
   437→                          <Avatar className="h-12 w-12 shrink-0">
   438→                            <AvatarFallback className="bg-primary text-primary-foreground text-sm font-bold">
   439→                              {(activeStay.guest.firstName || '?')[0]}{(activeStay.guest.lastName || '?')[0]}
   440→                            </AvatarFallback>
   441→                          </Avatar>
   442→                          <div className="space-y-1.5">
   443→                            <p className="text-sm font-semibold">
   444→                              {activeStay.guest.firstName} {activeStay.guest.lastName}
   445→                            </p>
   446→                            <p className="text-xs text-muted-foreground flex items-center gap-1.5">
   447→                              <Phone className="w-3 h-3 shrink-0" /> {activeStay.guest.phone}
   448→                            </p>
   449→                            {activeStay.guest.email && (
   450→                              <p className="text-xs text-muted-foreground flex items-center gap-1.5">
   451→                                <Mail className="w-3 h-3 shrink-0" /> {activeStay.guest.email}
   452→                              </p>
   453→                            )}
   454→                          </div>
   455→                        </div>
   456→                      </div>
   457→
   458→                      {/* Check-in / Check-out */}
   459→                      <div className="grid grid-cols-2 gap-4">
   460→                        <div className="bg-muted/40 rounded-lg p-4">
   461→                          <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider mb-2 flex items-center gap-1">
   462→                            <Calendar className="w-3 h-3" /> Check-in
   463→                          </p>
   464→                          <p className="text-sm font-medium">{activeStay.checkInDate}</p>
   465→                          <p className="text-xs text-muted-foreground mt-0.5">{activeStay.checkInTime}</p>
   466→                        </div>
   467→                        <div className="bg-muted/40 rounded-lg p-4">
   468→                          <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider mb-2 flex items-center gap-1">
   469→                            <Calendar className="w-3 h-3" /> Check-out
   470→                          </p>
   471→                          <p className="text-sm font-medium">{activeStay.checkOutDate}</p>
   472→                          <p className="text-xs text-muted-foreground mt-0.5">{activeStay.checkOutTime}</p>
   473→                        </div>
   474→                      </div>
   475→
   476→                      {/* Identification Details */}
   477→                      {(activeStay.guest.idNumber || activeStay.guest.dateOfBirth || activeStay.guest.idType) && (
   478→                        <div>
   479→                          <p className="text-[10px] text-muted-foreground uppercase font-semibold tracking-wider mb-3 flex items-center gap-1.5">
   480→                            <Shield className="w-3.5 h-3.5" /> Identification
   481→                          </p>
   482→                          <div className="bg-muted/40 rounded-lg p-4 space-y-3">
   483→                            {activeStay.guest.idType && (
   484→                              <div className="flex justify-between text-xs">
   485→                                <span className="text-muted-foreground">ID Type</span>
   486→                                <span className="font-medium">{activeStay.guest.idType}</span>
   487→                              </div>
   488→                            )}
   489→                            {activeStay.guest.idNumber && (
   490→                              <div className="flex justify-between text-xs">
   491→                                <span className="text-muted-foreground">ID Number</span>
   492→                                <span className="font-medium font-mono">{activeStay.guest.idNumber}</span>
   493→                              </div>
   494→                            )}
