    33→
    34→type IdMethod = 'manual' | 'phone-scan';
    35→
    36→export default function CheckIn() {
    37→  const rooms = useMotelStore((s) => s.rooms);
    38→  const addGuest = useMotelStore((s) => s.addGuest);
    39→  const addStay = useMotelStore((s) => s.addStay);
    40→  const addPayment = useMotelStore((s) => s.addPayment);
    41→  const updateRoomStatus = useMotelStore((s) => s.updateRoomStatus);
    42→  const addActivity = useMotelStore((s) => s.addActivity);
    43→  const setActiveTab = useMotelStore((s) => s.setActiveTab);
    44→
    45→  // Step & method
    46→  const [step, setStep] = useState(0);
    47→  const [idMethod, setIdMethod] = useState<IdMethod | null>(null);
    48→
    49→  // Room selection
    50→  const [roomType, setRoomType] = useState<RoomType>('1-bed');
    51→  const [selectedRoomId, setSelectedRoomId] = useState('');
    52→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('card');
    53→  const [termsAccepted, setTermsAccepted] = useState(false);
    54→
    55→  // Guest form fields
    56→  const [firstName, setFirstName] = useState('');
    57→  const [lastName, setLastName] = useState('');
    58→  const [phone, setPhone] = useState('');
    59→  const [email, setEmail] = useState('');
    60→  const [idNumber, setIdNumber] = useState('');
    61→  const [idType, setIdType] = useState('Driver License');
    62→  const [idState, setIdState] = useState('');
    63→  const [dateOfBirth, setDateOfBirth] = useState('');
    64→  const [checkInDate, setCheckInDate] = useState(() => new Date().toISOString().split('T')[0]);
    65→  const [checkOutDate, setCheckOutDate] = useState(() =>
    66→    new Date(Date.now() + 86400000).toISOString().split('T')[0]
    67→  );
    68→  const [checkInTime, setCheckInTime] = useState('2:00 PM');
    69→  const [checkOutTime, setCheckOutTime] = useState('10:00 AM');
    70→
    71→  // ID photo (manual upload)
    72→  const [idPhotoPreview, setIdPhotoPreview] = useState<string | null>(null);
    73→  const [idPhotoUrl, setIdPhotoUrl] = useState<string>('');
    74→  const fileInputRef = useRef<HTMLInputElement>(null);
    75→
    76→  // Signature
    77→  const sigCanvas = useRef<SignatureCanvas>(null);
    78→  const [signatureDataUrl, setSignatureDataUrl] = useState<string>('');
    79→

    82→  const [scanReceived, setScanReceived] = useState(false);
    83→  const [scannedImageUrl, setScannedImageUrl] = useState<string>('');
    84→  const [scannedSignatureUrl, setScannedSignatureUrl] = useState<string>('');

    86→
    87→  const [completedStayId, setCompletedStayId] = useState('');

    89→  // ── Derived ──
    90→  const availableRooms = useMemo(
    91→    () => rooms.filter((r) => r.status === 'available' && r.type === roomType),
    92→    [rooms, roomType]

    94→  const selectedRoom = rooms.find((r) => r.id === selectedRoomId);
    95→  const rate = selectedRoom?.rate ?? (roomType === '1-bed' ? 65 : 85);
    96→

    98→  const scanUrl = typeof window !== 'undefined'
    99→    ? `${window.location.origin}/scan/${scanSessionId}`
   100→    : '';
   101→

   106→    const channel = supabase.channel(`scan_${scanSessionId}`, {

   112→      if (imageUrl) {
   113→        setScannedImageUrl(imageUrl);
   114→        setIdPhotoUrl(imageUrl);
   115→      }
   116→      if (sigUrl) {
   117→        setScannedSignatureUrl(sigUrl);
   118→        setSignatureDataUrl(sigUrl);

   133→      if (supabase) supabase.removeChannel(channel);

   143→    // Preview
   144→    const reader = new FileReader();
   145→    reader.onload = (ev) => setIdPhotoPreview(ev.target?.result as string);

   148→    // Upload to Supabase Storage if connected

   150→      try {
   151→        const fileName = `id_admin_${Date.now()}_${file.name}`;
   152→        const { error } = await supabase.storage.from('ids').upload(fileName, file, {
   153→          contentType: file.type,
   154→          upsert: false,
   155→        });
   156→        if (error) {
   157→          toast.error('Failed to upload ID image to storage');
   158→          return;
   159→        }
   160→        const { data } = supabase.storage.from('ids').getPublicUrl(fileName);
   161→        setIdPhotoUrl(data.publicUrl);
   162→        toast.success('ID image uploaded');
   163→      } catch {
   164→        toast.error('Upload failed');
   165→      }
   166→    } else {
   167→      // No Supabase — just use the data URL as preview
   168→      setIdPhotoUrl(reader.result as string);
   169→    }

   181→      case 4: return true;
   182→      case 5:
   183→        if (idMethod === 'phone-scan') return termsAccepted && !!scannedSignatureUrl;
   184→        return termsAccepted && !!signatureDataUrl;
   185→      case 6: return false;
   186→      default: return false;
   187→    }
   188→  };

   190→  const handleNext = () => {
   191→    if (step < STEPS.length - 1) setStep(step + 1);
   192→  };

   194→  const handleBack = () => {
   195→    if (step > 0) setStep(step - 1);
   196→  };

   198→  const handleSubmit = () => {

   200→
   201→    const guestId = addGuest({
   202→      firstName,
   203→      lastName,
   204→      phone,
   205→      email,
   206→      idNumber,

   208→      idPhotoUrl: idPhotoUrl || undefined,
   209→      idType: idType || undefined,

   214→      guestId,
   215→      roomId: selectedRoomId,
   216→      checkInDate,
   217→      checkInTime,
   218→      checkOutDate,
   219→      checkOutTime,
   220→      rateAmount: rate,
   221→      status: 'active',
   222→      keyDeposit: 10,
   223→      tvRemoteDeposit: 10,

   226→    addPayment({
   227→      stayId,
   228→      amount: rate,
   229→      method: paymentMethod,
   230→      description: `Room charge (1 night, ${roomType})`,

   233→    addPayment({
   234→      stayId,
   235→      amount: 20,
   236→      method: paymentMethod,
   237→      description: 'Key + TV remote deposit',
   238→    });

   240→    updateRoomStatus(selectedRoomId, 'occupied');

   242→    addActivity({
   243→      guest: `${firstName} ${lastName}`,
   244→      action: 'Check-in',
   245→      room: selectedRoom.roomNumber,
   246→      time: new Date().toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: true }),
   247→      status: 'Success',
   248→    });
   249→
   250→    // Save signature to Supabase if connected
   251→    const sigData = idMethod === 'phone-scan' ? scannedSignatureUrl : signatureDataUrl;

   253→      supabase.from('signatures').insert({
   254→        stay_id: stayId,
   255→        guest_id: guestId,
   256→        signature_data_url: sigData,
   257→      }).then(({ error }) => {
   258→        if (error) console.error('Failed to save signature:', error);
   259→      });
   260→    }
   261→
   262→    setCompletedStayId(stayId);
   263→    setStep(6);
   264→    toast.success('Check-in completed successfully!');
   265→  };
   266→
   267→  const resetForm = () => {
   268→    setStep(0);
   269→    setIdMethod(null);
   270→    setRoomType('1-bed');
   271→    setSelectedRoomId('');
   272→    setPaymentMethod('card');
   273→    setTermsAccepted(false);
   274→    setFirstName('');
   275→    setLastName('');
   276→    setPhone('');
   277→    setEmail('');
   278→    setIdNumber('');
   279→    setIdType('Driver License');
   280→    setIdState('');
   281→    setDateOfBirth('');
   282→    setCheckInDate(new Date().toISOString().split('T')[0]);
   283→    setCheckOutDate(new Date(Date.now() + 86400000).toISOString().split('T')[0]);
   284→    setCheckInTime('2:00 PM');
   285→    setCheckOutTime('10:00 AM');
   286→    setIdPhotoPreview(null);
   287→    setIdPhotoUrl('');
   288→    setSignatureDataUrl('');
   289→    setScannedImageUrl('');
   290→    setScannedSignatureUrl('');
   291→    setScanReceived(false);
   292→    setCompletedStayId('');
   293→  };
   294→
   295→  return (
   296→    <div className="p-4 lg:p-6">
   297→      <div className="mb-6">
   298→        <h2 className="text-2xl font-bold text-foreground">New Check-In</h2>

   309→                  ? 'bg-primary text-primary-foreground'

   312→                    : 'bg-muted text-muted-foreground'

   315→              {i < step ? (
   316→                <CheckCircle className="w-3 h-3" />

   318→                <span>{i + 1}</span>
   319→              )}
   320→              <span className="hidden sm:inline">{label}</span>
   321→            </div>

   325→          </div>
   326→        ))}

   335→            <CardHeader>
   336→              <CardTitle className="text-lg">Select Room Type</CardTitle>

   339→              <div className="grid grid-cols-2 gap-4">
   340→                <button

   343→                    roomType === '1-bed'
   344→                      ? 'border-primary bg-primary/5'
   345→                      : 'border-border hover:border-primary/40'
   346→                  }`}

   348→                  <BedSingle className="w-8 h-8 mb-2 text-muted-foreground" />
   349→                  <p className="font-semibold">1-Bed King</p>
   350→                  <p className="text-sm text-muted-foreground">$65/night</p>
   351→                </button>

   353→                  onClick={() => { setRoomType('2-bed'); setSelectedRoomId(''); }}

   384→
   385→        {/* ── Step 1: ID Method Choice ── */}

   387→          <Card>
   388→            <CardHeader>
   389→              <CardTitle className="text-lg">How will you verify the guest&apos;s ID?</CardTitle>
   390→            </CardHeader>
   391→            <CardContent className="space-y-4">
   392→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   393→                {/* Manual Entry */}
   394→                <button
   395→                  onClick={() => setIdMethod('manual')}
   396→                  className={`p-6 rounded-lg border-2 transition-all cursor-pointer text-left space-y-3 ${
   397→                    idMethod === 'manual'
   398→                      ? 'border-primary bg-primary/5'
   399→                      : 'border-border hover:border-primary/40'
   400→                  }`}
   401→                >
   402→                  <div className="flex items-center gap-3">
   403→                    <Keyboard className="w-8 h-8 text-muted-foreground" />
   404→                    <span className="font-semibold text-lg">Manual Entry</span>

   406→                  <p className="text-sm text-muted-foreground">
   407→                    Type in ID details and upload a photo of the ID from this computer.
   408→                  </p>
   409→                  {idMethod === 'manual' && (
   410→                    <Badge className="bg-primary text-primary-foreground">Selected</Badge>

   412→                </button>
   413→
   414→                {/* Phone Scan */}

   416→                  onClick={() => setIdMethod('phone-scan')}

   418→                    idMethod === 'phone-scan'
   419→                      ? 'border-primary bg-primary/5'
   420→                      : 'border-border hover:border-primary/40'
   421→                  }`}

   423→                  <div className="flex items-center gap-3">

   427→                  <p className="text-sm text-muted-foreground">
   428→                    Show a QR code to the guest. They scan it with their phone camera to upload their ID photo and sign.
   429→                  </p>
   430→                  {idMethod === 'phone-scan' && (
   431→                    <Badge className="bg-primary text-primary-foreground">Selected</Badge>
   432→                  )}
   433→                </button>
   434→              </div>
   435→            </CardContent>
   436→          </Card>
   437→        )}
   438→
   439→        {/* ── Step 2: Guest Details ── */}

   442→            <CardHeader>
   443→              <CardTitle className="text-lg">Guest Details</CardTitle>

   450→                </div>

   454→                </div>
   455→              </div>

   459→              </div>

   461→                <Label htmlFor="email" className="mb-1.5 block">Email</Label>

   464→              <div className="grid grid-cols-2 gap-4">

   491→                      {['10:00 AM', '11:00 AM', '12:00 PM'].map((t) => (
   492→                        <SelectItem key={t} value={t}>{t}</SelectItem>
   493→                      ))}
   494→                    </SelectContent>
   495→                  </Select>
   496→                </div>
   497→              </div>
   498→            </CardContent>
   499→          </Card>

   501→
   502→        {/* ── Step 3: ID Info / Scan ── */}
